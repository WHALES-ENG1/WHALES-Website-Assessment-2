package com.mathochist.mazegame.entity.npc;

import com.badlogic.gdx.Gdx;
import com.badlogic.gdx.graphics.Texture;
import com.badlogic.gdx.graphics.g2d.Animation;
import com.badlogic.gdx.graphics.g2d.Sprite;
import com.badlogic.gdx.graphics.g2d.SpriteBatch;
import com.badlogic.gdx.graphics.g2d.TextureAtlas;
import com.badlogic.gdx.graphics.g2d.TextureRegion;
import com.mathochist.mazegame.Main;
import com.mathochist.mazegame.entity.MapEntity;
import com.mathochist.mazegame.entity.Player;
import com.mathochist.mazegame.render.RenderBuffer;
import com.mathochist.mazegame.render.RenderObject;
import com.mathochist.mazegame.userinterface.Hud;
import com.mathochist.mazegame.userinterface.speechtext.SpeechType;
import com.mathochist.mazegame.worlds.GameWorld;

/**
 * An NPC that detects if the player is within range and chases the player indefinitely.
 *
 * @author Hamza Ahmad Dean
 */
public class Dean extends MapEntity implements Inpc {

  public static final float SPRITE_WIDTH = 32.0f;
  public static final float SPRITE_HEIGHT = 52.0f;
  public static final float DETECTION_RANGE = 1280f; // pixels
  private static final float CHASE_SPEED = 300.0f; // pixels per second
  public static final float CATCH_DISTANCE = 64f; // pixels
  public static final float CATCH_PENALTY = 10;
  private static final int FRAME_COLS = 3;
  private static final int FRAME_ROWS = 4;
  private static final int IDLE_FRAME_INDEX = 1;
  private final Texture spriteSheet;
  private final Animation<TextureRegion> walkUp;
  private final Animation<TextureRegion> walkDown;
  private final Animation<TextureRegion> walkLeft;
  private final Animation<TextureRegion> walkRight;
  private final TextureRegion idleUp;
  private final TextureRegion idleDown;
  private final TextureRegion idleLeft;
  private final TextureRegion idleRight;
  private final Sprite deanSprite;
  private TextureRegion currentFrame;
  private float currentX;
  private float currentY;
  private boolean isChasing = false;
  private float stateTime = 0f;
  private double timeSinceLastPenalty = 0;
  private boolean positionInitialised = false;
  private int direction = 0; // 0=down,1=left,2=right,3=up

  /**
   * Create Dean.
   *
   * @param game Current Game instance
   * @param batch SpriteBatch
   * @param atlas Atlas containing dean
   * @param tileX dean spawn X coord
   * @param tileY dean spawn Y coord
   * @param width dean width
   * @param height dean height
   * @param collidable is dean collidable
   */
  public Dean(
      Main game,
      SpriteBatch batch,
      TextureAtlas atlas,
      Integer tileX,
      Integer tileY,
      Integer width,
      Integer height,
      Boolean collidable) {
    super(game, batch, atlas, "dean", tileX, tileY, width, height, collidable);

    // Load Dean's sprite sheet
    spriteSheet = new Texture(Gdx.files.internal("maps/library/Temp Dean animations.png"));
    TextureRegion[][] tmp =
        TextureRegion.split(
            spriteSheet, spriteSheet.getWidth() / FRAME_COLS, spriteSheet.getHeight() / FRAME_ROWS);

    // Create animations for each direction
    walkUp = new Animation<>(0.15f, tmp[3]);
    walkLeft = new Animation<>(0.15f, tmp[1]);
    walkDown = new Animation<>(0.15f, tmp[0]);
    walkRight = new Animation<>(0.15f, tmp[2]);

    // Start with first frame facing down
    currentFrame = tmp[0][1];

    deanSprite = new Sprite(spriteSheet);

    // Initialise position based on tile coordinates
    this.currentX = tileX * SPRITE_WIDTH;
    this.currentY = tileY * SPRITE_HEIGHT;

    idleUp = tmp[3][IDLE_FRAME_INDEX];
    idleDown = tmp[0][IDLE_FRAME_INDEX];
    idleLeft = tmp[1][IDLE_FRAME_INDEX];
    idleRight = tmp[2][IDLE_FRAME_INDEX];
  }

  /**
   * Updates the NPC's position and chasing behavior. Called from GameWorld's update loop.
   *
   * @param delta the time since the last frame in seconds
   * @param player the player entity
   * @param world the game world containing collision and tile data
   */
  public void update(float delta, Player player, GameWorld world) {
    stateTime += delta;

    // Initialise proper pixel position on first update when GameWorld is available
    if (!positionInitialised) {
      float[] startPos = world.getPixels((int) getTileX(), (int) getTileY());
      this.currentX = startPos[0];
      this.currentY = startPos[1];
      positionInitialised = true;
    }

    float playerX = player.getX();
    float playerY = player.getY();

    // Calculate distance to player
    float distanceX = playerX - currentX;
    float distanceY = playerY - currentY;
    double distance = Math.sqrt(distanceX * distanceX + distanceY * distanceY);

    // Check if player enters detection range
    isChasing = distance <= DETECTION_RANGE;

    // Handle chase logic
    if (isChasing) {

      // Check if player has been caught
      if (distance <= CATCH_DISTANCE) {
        onCatchPlayer(world);
        return;
      }

      // Move toward player
      float angle = (float) Math.atan2(playerY - currentY, playerX - currentX);
      float moveX = (float) Math.cos(angle) * CHASE_SPEED * delta;
      float moveY = (float) Math.sin(angle) * CHASE_SPEED * delta;

      // Check collision before moving
      float newX = currentX + moveX;
      float newY = currentY + moveY;

      boolean[] collisionLayer = world.getCollisionLayer(newX, newY);

      // Collision check, stop if any collision detected
      boolean canMove = true;
      if (moveY > 0 && collisionLayer[0]) {
        canMove = false; // up
      }
      if (moveY < 0 && collisionLayer[1]) {
        canMove = false; // down
      }
      if (moveX < 0 && collisionLayer[2]) {
        canMove = false; // left
      }
      if (moveX > 0 && collisionLayer[3]) {
        canMove = false; // right
      }

      if (canMove) {
        currentX = newX;
        currentY = newY;

        // Update animation based on movement direction
        if (Math.abs(moveY) > Math.abs(moveX)) {
          // Moving more vertically
          if (moveY > 0) {
            currentFrame = walkUp.getKeyFrame(stateTime, true);
            direction = 3;
          } else {
            currentFrame = walkDown.getKeyFrame(stateTime, true);
            direction = 0;
          }
        } else {
          // Moving more horizontally
          if (moveX < 0) {
            currentFrame = walkLeft.getKeyFrame(stateTime, true);
            direction = 1;
          } else {
            currentFrame = walkRight.getKeyFrame(stateTime, true);
            direction = 2;
          }
        }
      } else {
        // Can't move, show idle frame
        stateTime = 0;
        switch (direction) {
          case 3 -> currentFrame = idleUp;
          case 0 -> currentFrame = idleDown;
          case 1 -> currentFrame = idleLeft;
          case 2 -> currentFrame = idleRight;
          default -> throw new IllegalArgumentException("Unknown Enum value.");
        }
      }
    } else {
      // Show idle frame when not chasing
      stateTime = 0;
      switch (direction) {
        case 3 -> currentFrame = idleUp;
        case 0 -> currentFrame = idleDown;
        case 1 -> currentFrame = idleLeft;
        case 2 -> currentFrame = idleRight;
        default -> throw new IllegalArgumentException("Unknown Enum value.");
      }
    }

    // Update sprite region
    deanSprite.setRegion(currentFrame);
  }

  /**
   * Called when Dean catches the player.
   *
   * @param world the game world containing collision and tile data
   */
  private void onCatchPlayer(GameWorld world) {
    world
        .parentScreen
        .getGameHud()
        .getSpeechBubbleManager()
        .removeBubblesOfType(SpeechType.NPC_SPEECH);
    world
        .parentScreen
        .getGameHud()
        .getSpeechBubbleManager()
        .createSpeechBubble(SpeechType.NPC_SPEECH, "Dean: GOT YOU!", 2000);
    if (timeSinceLastPenalty == 0 || (System.currentTimeMillis() - timeSinceLastPenalty) >= 1000) {
      game.getTimerManager().addPenalty(CATCH_PENALTY);
      timeSinceLastPenalty = System.currentTimeMillis();
    }
    game.getEventsCounter().deanCatchEvent();
  }

  @Override
  public void render(GameWorld world, RenderBuffer buffer) {
    // Use current pixel position
    deanSprite.setPosition((int) currentX, (int) currentY);

    // Calculate render layer based on Y position
    float[] tilePos = world.pixelCoordsToTileIndex(currentX, currentY);
    int renderLayer = (int) Math.ceil(tilePos[1]); // Layer based on Y position

    buffer.addToBuffer(new RenderObject(deanSprite, super.getBatch(), renderLayer));
  }

  @Override
  public void renderPrompt(GameWorld world) {}

  @Override
  public void renderSpeech(String text, GameWorld world) {}

  @Override
  public boolean onInteract(Player p, GameWorld world, Hud worldHud) {
    return true;
  }

  @Override
  public void onInteractEnd(Player p, GameWorld world) {}

  @Override
  public boolean onCollision(GameWorld world) {
    return true; // Handle collision
  }

  /** dispose Used assets. */
  public void dispose() {
    spriteSheet.dispose();
  }

  public boolean isChasing() {
    return isChasing;
  }

  public float getCurrentX() {
    return currentX;
  }

  public float getCurrentY() {
    return currentY;
  }
}
