package com.mathochist.mazegame.screen.roomscreen;

import com.mathochist.mazegame.Main;
import com.mathochist.mazegame.userinterface.speechtext.SpeechType;

// CHANGED: Added this class to remove duplication from library screens that all had the same
// functionality with
// the sprint event.

/** A game screen in which running triggers the Library Sprint Event. */
public abstract class BaseLibraryScreen extends RoomScreen {
  private static final int SPRINT_PENALTY = 5;
  private double timeSinceLastSprintPenalty = 0;

  /**
   * Use other Constructor.
   *
   * @param game current instance of game
   * @param mapFileName Name of room
   */
  public BaseLibraryScreen(Main game, String mapFileName) {
    this(game, null, null, mapFileName);
  }

  /**
   * Create room using Room Screen.
   *
   * @param game current instance of game
   * @param spawnX X coord to spawn player at
   * @param spawnY Y coord to spawn player at
   * @param mapFileName Name of room
   */
  public BaseLibraryScreen(Main game, Float spawnX, Float spawnY, String mapFileName) {
    super(game, spawnX, spawnY, mapFileName);
  }

  @Override
  public void render(float delta) {
    super.render(delta);

    // Check if the player is sprinting and trigger the negative event if they are
    if (super.getPlayer().isSprinting()) {
      super.getGameHud().getSpeechBubbleManager().removeBubblesOfType(SpeechType.NPC_SPEECH);
      super.getGameHud()
          .getSpeechBubbleManager()
          .createSpeechBubble(SpeechType.NPC_SPEECH, "Librarian: Oi, stop running!", 1000);
      if (timeSinceLastSprintPenalty == 0
          || (System.currentTimeMillis() - timeSinceLastSprintPenalty) >= 1000) {
        game.getTimerManager().addPenalty(SPRINT_PENALTY);
        timeSinceLastSprintPenalty = System.currentTimeMillis();
      }
      game.getEventsCounter().librarySprintEvent();
    }
  }

  @Override
  public void show() {
    super.show();
  }
}
