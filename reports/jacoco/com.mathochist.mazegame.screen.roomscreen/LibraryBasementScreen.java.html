package com.mathochist.mazegame.screen.roomscreen;

import com.mathochist.mazegame.Main;

/**
 * Create Library Basement.
 */
public class LibraryBasementScreen extends RoomScreen {
  private static final String MAP_FILE_NAME = "maps/library_basement.json";

  /**
   * Use other Constructor.
   *
   * @param game current instance of game
   */
  public LibraryBasementScreen(Main game) {
    this(game, null, null);
  }

  /**
   * Create Library basement using Room Screen.
   *
   * @param game current instance of game
   * @param spawnX X coord to spawn at
   * @param spawnY Y coord to spawn at
   */
  public LibraryBasementScreen(Main game, Float spawnX, Float spawnY) {
    super(game, spawnX, spawnY, MAP_FILE_NAME);
    game.getEventsCounter().fallenIntoBasementEvent(); // Increment hidden event counter
    super.setVignette(VignetteLevel.dark);
  }
}
