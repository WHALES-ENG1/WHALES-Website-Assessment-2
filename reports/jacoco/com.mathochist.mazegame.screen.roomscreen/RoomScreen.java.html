package com.mathochist.mazegame.screen.roomscreen;

import com.badlogic.gdx.Gdx;
import com.badlogic.gdx.Input;
import com.badlogic.gdx.audio.Music;
import com.badlogic.gdx.audio.Sound;
import com.badlogic.gdx.graphics.Cursor;
import com.badlogic.gdx.graphics.GL20;
import com.badlogic.gdx.graphics.OrthographicCamera;
import com.badlogic.gdx.graphics.Texture;
import com.badlogic.gdx.graphics.g2d.SpriteBatch;
import com.badlogic.gdx.utils.viewport.FitViewport;
import com.mathochist.mazegame.Main;
import com.mathochist.mazegame.entity.Player;
import com.mathochist.mazegame.entity.playerinventory.InventoryObject;
import com.mathochist.mazegame.render.RenderBuffer;
import com.mathochist.mazegame.screen.AchievementScreen;
import com.mathochist.mazegame.screen.DefaultScreen;
import com.mathochist.mazegame.screen.EndScreen;
import com.mathochist.mazegame.screen.UsernameScreen;
import com.mathochist.mazegame.userinterface.Hud;
import com.mathochist.mazegame.userinterface.speechtext.SpeechType;
import com.mathochist.mazegame.worlds.GameWorld;

/**
 * Base class for all game screens, providing common functionality such as camera setup, viewport
 * management, HUD integration, and player handling.
 */
public abstract class RoomScreen extends DefaultScreen {
  public static boolean isFireAlarmActive = false;
  public static boolean powerOutage = false;
  public static boolean fireEventCheck = true;
  public static double fireAlarmLuck = Math.random();
  public final Music fireAlarmSound;
  private final OrthographicCamera camera;
  private final FitViewport viewport;
  private final Hud gameHud;
  private final SpriteBatch screenBatch;
  private final GameWorld world;
  private final RenderBuffer renderBuffer;
  private final Texture vignetteLight;
  private final Texture vignetteMedium;
  private final Texture vignetteDark;
  private final Texture vignetteFire;
  private float outageTimer = 20f; // power goes out for 20 seconds
  private float fireAlarmTimer = 10f; // fire alarm time
  private Player player;
  private VignetteLevel currentVignette = VignetteLevel.medium;
  private boolean isPaused = false;
  private boolean prepareTransition = false;
  private float inputDelay = 0.0f;

  /**
   * Create Room Screen using the other Constructor.
   *
   * @param game current instance of game
   * @param mapFileName Name of room
   */
  public RoomScreen(Main game, String mapFileName) {
    this(game, null, null, mapFileName);
  }

  /**
   * Create the Room Screen to display.
   *
   * @param game Current instance of game
   * @param spawnX X coordinate to place the player at
   * @param spawnY Y coordinate to place the player at
   * @param mapFileName Name of room
   */
  public RoomScreen(Main game, Float spawnX, Float spawnY, String mapFileName) {
    super(game);
    // GDX setup
    Gdx.graphics.setSystemCursor(Cursor.SystemCursor.Arrow);
    Gdx.input.setInputProcessor(this);

    // Camera setup
    camera = new OrthographicCamera();
    float gameWidth = 1280;
    float gameHeight = 720;
    viewport = new FitViewport(gameWidth, gameHeight, camera);
    camera.position.set(gameWidth / 2f, gameHeight / 2f, 0); // Center camera
    camera.update();

    gameHud = new Hud(game);
    // Batch setup
    screenBatch = game.getSpriteBatch();
    // Render buffer setup
    renderBuffer = new RenderBuffer();

    // CHANGE: vignette setup
    vignetteLight = new Texture(Gdx.files.internal("vignette/light.png"));
    vignetteMedium = new Texture(Gdx.files.internal("vignette/medium.png"));
    vignetteDark = new Texture(Gdx.files.internal("vignette/dark.png"));
    vignetteFire = new Texture(Gdx.files.internal("vignette/fire-alarm.png"));

    vignetteLight.setFilter(Texture.TextureFilter.Linear, Texture.TextureFilter.Linear);
    vignetteMedium.setFilter(Texture.TextureFilter.Linear, Texture.TextureFilter.Linear);
    vignetteDark.setFilter(Texture.TextureFilter.Linear, Texture.TextureFilter.Linear);
    this.world = new GameWorld(game, this, Gdx.files.internal(mapFileName), screenBatch);
    String currentMap = mapFileName.replace("maps/", "").replace(".json", "").replace("_", " ");
    currentMap = currentMap.substring(0, 1).toUpperCase() + currentMap.substring(1);

    float[] spawnPixels = getSpawnPixels(spawnX, spawnY);

    camera.position.set(spawnPixels[0], spawnPixels[1], 0);
    camera.update();

    this.player = new Player(game, camera, screenBatch, world, spawnPixels[0], spawnPixels[1]);

    if (game.getPlayerInventory().hasItem(InventoryObject.GLASSES)) {
      camera.zoom = 1.5f;
      camera.update();
    }

    getGameHud().getSpeechBubbleManager().removeBubblesOfType(SpeechType.NPC_SPEECH);
    getGameHud()
        .getSpeechBubbleManager()
        .createSpeechBubble(SpeechType.NPC_SPEECH, currentMap, 1000);

    // Setup sounds
    fireAlarmSound = Gdx.audio.newMusic(Gdx.files.internal("audio/fire_alarm.mp3"));
  }

  /**
   * Calculate the pixels to spawn the player at.
   *
   * @param spawnX Specified spawn X position or null if unspecified
   * @param spawnY Specified spawn Y position or null if unspecified
   * @return An array with 2 items, the X pixel pos and the Y pixel pos
   */
  private float[] getSpawnPixels(Float spawnX, Float spawnY) {
    float[] spawnPixels;
    if (spawnX == null || spawnY == null) {
      // If null, grab from the world definition
      int[] spawnPoint = world.getSpawnPointPixels();
      spawnPixels = new float[] {spawnPoint[0], spawnPoint[1]};
    } else {
      // Otherwise use the specified coordinates from the parameters
      spawnPixels = world.getPixels(spawnX, spawnY);
    }
    return spawnPixels;
  }

  public Player getPlayer() {
    return player;
  }

  public void setPlayer(Player player) {
    this.player = player;
  }

  @Override
  public void show() {
    Gdx.input.setInputProcessor(this);
    isPaused = false;
    inputDelay = 0.2f;
    if (player != null) {
      player.getKeyBuffer().clear();
    }
  }

  @Override
  public void hide() {
    isPaused = true;
  }

  @Override
  public void render(float delta) {
    Gdx.gl.glClearColor(0, 0, 0, 1);
    Gdx.gl.glClear(GL20.GL_COLOR_BUFFER_BIT);
    if (prepareTransition) {
      return;
    }
    if (inputDelay > 0.0f) {
      inputDelay -= delta;
    } else {
      if (Gdx.input.isKeyJustPressed(Input.Keys.TAB)) {
        game.setScreen(new AchievementScreen(game, this));
        return;
      }

      if (Gdx.input.isKeyJustPressed(Input.Keys.ESCAPE)
          || Gdx.input.isKeyJustPressed(Input.Keys.P)) {
        isPaused = !isPaused;
        if (!isPaused) {
          Gdx.input.setInputProcessor(this);
        }
      }
    }
    if (isPaused) {
      if (gameHud != null) {
        gameHud.renderPauseScreen();
      }
      return;
    }
    camera.update();
    if (world != null) {
      world.render(viewport, renderBuffer);
    }
    if (player != null) {
      player.update(delta, renderBuffer);
    }
    if (world != null) {
      world.update(delta, player);
    }

    screenBatch.setProjectionMatrix(camera.combined);
    screenBatch.begin();

    if (renderBuffer != null) {
      for (var obj : renderBuffer.getBufferOrderedByZaxisIndex()) {
        obj.render();
      }
      renderBuffer.clearBuffer();
    }
    screenBatch.end();

    handleFireAlarmLogic(delta);
    handlePowerOutageLogic(delta);

    renderVignette(currentVignette);
    if (gameHud != null) {
      gameHud.render();
    }
    if (game.getTimerManager().isTimeUp()) {
      endGame();
    }
  }

  private void endGame() {
    if (world != null) {
      world.getBackgroundMusic().stop();
    }

    if (game.getUsername() == null) {
      game.setScreen(new UsernameScreen(game));
    } else {
      game.setScreen(new EndScreen(game, null));
    }
  }

  private void handlePowerOutageLogic(float delta) {
    if (powerOutage) {
      this.currentVignette = VignetteLevel.dark;
      outageTimer -= delta;
    }
    if (outageTimer <= 0.0f) {
      powerOutage = false;
      this.currentVignette = VignetteLevel.medium;
    }
  }

  private void handleFireAlarmLogic(float delta) {
    // FireAlarmEvent Logic
    if (game.getTimerManager().getTimeLeft() <= 120) {
      if (fireEventCheck) {
        fireEventCheck = false;
        if (fireAlarmLuck > 0.5) {
          isFireAlarmActive = true;
          if (world != null) {
            world.getBackgroundMusic().stop();
          }
          game.getEventsCounter().fireEvent();
          dispose();
          MorrellScreen screen = new MorrellScreen(game);
          screen.getGameHud().getSpeechBubbleManager().clear();
          screen
              .getGameHud()
              .getSpeechBubbleManager()
              .createSpeechBubble(SpeechType.NPC_SPEECH, "The fire alarm went off!", 5000);
          screen.getWorld().setBackgroundMusic(fireAlarmSound);
          game.setScreen(screen);
          return;
        }
      }
    }
    if (isFireAlarmActive && fireAlarmTimer >= 0) {
      this.currentVignette = VignetteLevel.fire;
      fireAlarmTimer -= delta;
    } else if (isFireAlarmActive) {
      this.currentVignette = VignetteLevel.medium;
      isFireAlarmActive = false;
      world.setBackgroundMusic(Gdx.audio.newMusic(Gdx.files.internal("audio/library_theme.mp3")));
    }
  }

  @Override
  public void renderShapes() {
    game.getShapeRenderer().begin();
    world.renderDebugLayerShapes();
    game.getShapeRenderer().end();
  }

  @Override
  public void resize(int width, int height) {
    viewport.update(width, height, false);
    if (gameHud != null) {
      gameHud.resize(width, height);
    }
  }

  public boolean isPrepareTransition() {
    return prepareTransition;
  }

  public void setPrepareTransition(boolean prepareTransition) {
    this.prepareTransition = prepareTransition;
  }

  public Hud getGameHud() {
    return gameHud;
  }

  // Input Processing methods can be added here if needed
  @Override
  public boolean keyDown(int keycode) {
    player.getKeyBuffer().addKeyPressed(keycode);
    return true;
  }

  @Override
  public boolean keyUp(int keycode) {
    player.getKeyBuffer().removeKeyPressed(keycode);
    return true;
  }

  /**
   * Render the vignette.
   *
   * @param level what level of vignette
   */
  public void renderVignette(VignetteLevel level) {
    Texture currentVignette;
    switch (level) {
      case light -> currentVignette = vignetteLight;
      case dark -> currentVignette = vignetteDark;
      case fire -> currentVignette = vignetteFire;
      default -> currentVignette = vignetteMedium;
    }
    screenBatch.setProjectionMatrix(screenBatch.getProjectionMatrix().idt());
    screenBatch.begin();
    screenBatch.draw(currentVignette, -1, -1, 2, 2);
    screenBatch.end();
    screenBatch.setProjectionMatrix(camera.combined);
  }

  public GameWorld getWorld() {
    return this.world;
  }

  public RenderBuffer getRenderBuffer() {
    return renderBuffer;
  }

  public boolean isPaused() {
    return isPaused;
  }

  public void setPaused(boolean paused) {
    isPaused = paused;
  }

  @Override
  public void dispose() {
    if (vignetteLight != null) {
      vignetteLight.dispose();
    }
    if (vignetteMedium != null) {
      vignetteMedium.dispose();
    }
    if (vignetteDark != null) {
      vignetteDark.dispose();
    }
    if (gameHud != null) {
      gameHud.dispose();
    }
    if (world != null) {
      world.dispose();
    }
    if (player != null) {
      player.dispose();
    }
    if (fireAlarmSound != null) {
      fireAlarmSound.dispose();
    }
  }

  public VignetteLevel getVignette() {
    return this.currentVignette;
  }

  public void setVignette(VignetteLevel type) {
    this.currentVignette = type;
  }

  public OrthographicCamera getCamera() {
    return camera;
  }

  /** Helper method to set fire alarm active during tests. * */
  public void setIsFireAlarmActive(boolean bool) {
    isFireAlarmActive = bool;
  }
}
