package com.mathochist.mazegame.screen.roomscreen;

import com.mathochist.mazegame.Main;

/** Create Burton. */
public class BurtonScreen extends BaseLibraryScreen {
  private static final String MAP_FILE_NAME = "maps/burton.json";

  /**
   * Use other Constructor.
   *
   * @param game current game instance
   */
  public BurtonScreen(Main game) {
    this(game, null, null);
  }

  /**
   * Create Burton using Room Screen.
   *
   * @param game current instance of game
   * @param spawnX X coord to spawn at
   * @param spawnY Y coord to spawn at
   */
  public BurtonScreen(Main game, Float spawnX, Float spawnY) {
    super(game, spawnX, spawnY, MAP_FILE_NAME);
  }
}
