package com.mathochist.mazegame;

import com.badlogic.gdx.Gdx;
import com.badlogic.gdx.graphics.g2d.SpriteBatch;
import com.badlogic.gdx.graphics.glutils.ShapeRenderer;
import com.mathochist.mazegame.entity.playerinventory.Inventory;
import com.mathochist.mazegame.screen.TitleScreen;
import com.mathochist.mazegame.screen.roomscreen.RoomScreen;
import com.mathochist.mazegame.userinterface.AchievementHandler;
import com.mathochist.mazegame.userinterface.EventsCounter;
import com.mathochist.mazegame.userinterface.ScoreCounter;
import com.mathochist.mazegame.userinterface.timers.TimerManager;

/** The Main class serves as the entry point for the game application. */
public class Main extends AllNighterGame {

  public int width = 0;
  public int height = 0;

  private TimerManager timerManager;
  private Inventory playerInventory;
  private EventsCounter eventsCounter;
  private AchievementHandler achievementHandler;
  private ScoreCounter scoreCounter;
  // CHANGED: Using one sprite batch for the game instead of many to make mocking for testing
  // easier.
  private SpriteBatch spriteBatch;
  // CHANGED: Using one shape renderer for the game instead of many as part of sprite batch
  // centralisation.
  private ShapeRenderer shapeRenderer;
  private String username;

  /** Initializes the game by setting the initial screen to the TitleScreen. */
  @Override
  public void create() {
    // Set title for the application window
    Gdx.graphics.setTitle("All Nighter");

    // Set aspect ratio to 16:9
    width = Gdx.graphics.getWidth() * 2;
    height = (width * 9) / 16;
    Gdx.graphics.setWindowedMode(width, height);

    this.spriteBatch = new SpriteBatch();
    this.shapeRenderer = new ShapeRenderer();
    this.shapeRenderer.setAutoShapeType(true);

    timerManager = new TimerManager();
    playerInventory = new Inventory();
    eventsCounter = new EventsCounter(this);
    scoreCounter = new ScoreCounter(this);
    achievementHandler = new AchievementHandler(this);

    // Set the initial screen to the title screen
    setScreen(new TitleScreen(this));
  }

  /** Disposes of resources and exits the application when the game is closed. */
  @Override
  public void dispose() {
    super.dispose();
    Gdx.app.exit();
  }

  /** reset game after playing. */
  public void resetGame() {
    this.timerManager = new TimerManager();
    this.playerInventory = new Inventory();
    this.eventsCounter = new EventsCounter(this);
    this.scoreCounter = new ScoreCounter(this);
    this.achievementHandler = new AchievementHandler(this);

    RoomScreen.isFireAlarmActive = false;
    RoomScreen.powerOutage = false;
    RoomScreen.fireEventCheck = true;
  }

  public TimerManager getTimerManager() {
    return timerManager;
  }

  public Inventory getPlayerInventory() {
    return playerInventory;
  }

  public EventsCounter getEventsCounter() {
    return eventsCounter;
  }

  public AchievementHandler getAchievementHandler() {
    return achievementHandler;
  }

  public ScoreCounter getScoreCounter() {
    return scoreCounter;
  }

  public SpriteBatch getSpriteBatch() {
    return spriteBatch;
  }

  public ShapeRenderer getShapeRenderer() {
    return shapeRenderer;
  }

  public String getUsername() {
    return username;
  }

  public void setUsername(String username) {
    this.username = username;
  }
}
