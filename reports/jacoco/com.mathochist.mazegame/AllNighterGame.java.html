package com.mathochist.mazegame;

import com.badlogic.gdx.Game;
import com.badlogic.gdx.Gdx;
import com.badlogic.gdx.Screen;
import com.mathochist.mazegame.screen.IbaseScreen;

/**
 * Abstract Game class that controls screen operations.
 */
public abstract class AllNighterGame extends Game {
  protected IbaseScreen screen;

  @Override
  public IbaseScreen getScreen() {
    return screen;
  }

  /**
   * Sets the current screen. {@link Screen#hide()} is called on any old screen, and {@link
   * Screen#show()} is called on the new screen, if any.
   *
   * @param screen may be {@code null}
   */
  public void setScreen(IbaseScreen screen) {
    if (this.screen != null) {
      this.screen.hide();
    }
    this.screen = screen;
    if (this.screen != null) {
      this.screen.show();
      this.screen.resize(Gdx.graphics.getWidth(), Gdx.graphics.getHeight());
    }
  }

  // if game receives a Screen
  @Override
  public void setScreen(Screen screen) {
    if (screen instanceof IbaseScreen) {
      setScreen((IbaseScreen) screen);
    } else {
      super.setScreen(screen);
    }
  }

  @Override
  public void render() {
    if (screen != null) {
      screen.render(Gdx.graphics.getDeltaTime());
      screen.renderShapes();
    }
  }

  @Override
  public void resize(int width, int height) {
    if (screen != null) {
      screen.resize(width, height);
    }
  }

  @Override
  public void pause() {
    if (screen != null) {
      screen.pause();
    }
  }

  @Override
  public void resume() {
    if (screen != null) {
      screen.resume();
    }
  }

  @Override
  public void dispose() {
    if (screen != null) {
      screen.dispose();
    }
  }
}
