package com.mathochist.mazegame.render;

import com.badlogic.gdx.graphics.g2d.Sprite;
import com.badlogic.gdx.graphics.g2d.SpriteBatch;

/** Define object to render. */
public class RenderObject {

  private Sprite sprite;
  private SpriteBatch batch;
  private int zaxisIndex;

  /**
   * Create Object to render.
   *
   * @param sprite Sprite to use
   * @param batch Sprite Batch
   * @param zaxisIndex "depth" of object
   */
  public RenderObject(Sprite sprite, SpriteBatch batch, int zaxisIndex) {
    this.sprite = sprite;
    this.batch = batch;
    this.zaxisIndex = zaxisIndex;
  }

  /** render by drawing the batch. */
  public void render() {
    sprite.draw(batch);
  }

  public Sprite getSprite() {
    return sprite;
  }

  public void setSprite(Sprite sprite) {
    this.sprite = sprite;
  }

  public SpriteBatch getBatch() {
    return batch;
  }

  public void setBatch(SpriteBatch batch) {
    this.batch = batch;
  }

  /**
   * get Z index.
   *
   * @return z Axis Index
   */
  public int getzIndex() {
    return zaxisIndex;
  }

  /**
   * Set Z index.
   *
   * @param zaxisIndex new Z index
   */
  public void setzIndex(int zaxisIndex) {
    this.zaxisIndex = zaxisIndex;
  }

  @Override
  public String toString() {
    return "RenderObject{"
        + "sprite="
        + sprite
        + ", batch="
        + batch
        + ", zIndex="
        + zaxisIndex
        + '}';
  }
}
