package com.mathochist.mazegame.worlds.object;

import com.badlogic.gdx.Gdx;
import com.badlogic.gdx.audio.Sound;
import com.badlogic.gdx.graphics.g2d.SpriteBatch;
import com.badlogic.gdx.graphics.g2d.TextureAtlas;
import com.mathochist.mazegame.Main;
import com.mathochist.mazegame.entity.MapEntity;
import com.mathochist.mazegame.entity.Player;
import com.mathochist.mazegame.entity.playerinventory.InventoryObject;
import com.mathochist.mazegame.render.RenderBuffer;
import com.mathochist.mazegame.userinterface.Hud;
import com.mathochist.mazegame.worlds.ExitConditions;
import com.mathochist.mazegame.worlds.ExitTile;
import com.mathochist.mazegame.worlds.GameWorld;

/**
 * Represents a hole in the library floor that leads to the basement. The hole is only visible and
 * collidable if the player has the keycard item due to game timeline.
 */
public class LibraryFloorHole extends MapEntity {

  private boolean visible;
  private Sound fallSound;

  /**
   * Creates the library floor hole object using Entity.
   *
   * @param game The instance of the game.
   * @param batch The current SpritBatch.
   * @param atlas Which Atlas object is in.
   * @param tileX X Coord to place at.
   * @param tileY Y Coord to place at.
   * @param width Width of object.
   * @param height Height of object.
   * @param collidable Is object collidable.
   */
  public LibraryFloorHole(
      Main game,
      SpriteBatch batch,
      TextureAtlas atlas,
      Integer tileX,
      Integer tileY,
      Integer width,
      Integer height,
      Boolean collidable) {
    super(game, batch, atlas, "hole_floor", tileX, tileY, width, height, collidable);

    // Visible if player has keycard
    visible = game.getPlayerInventory().hasItem(InventoryObject.KEYCARD);

    // Sound of falling into the basement
    fallSound = Gdx.audio.newSound(Gdx.files.internal("audio/crash_hurt.mp3"));
  }

  @Override
  public boolean onInteract(Player p, GameWorld world, Hud worldHud) {
    return false; // No interaction behavior
  }

  @Override
  public void onInteractEnd(Player p, GameWorld world) {} // No action needed on interaction end

  @Override
  public void render(GameWorld world, RenderBuffer renderBuffer) {

    if (!visible) {
      return;
    }
    super.render(world, renderBuffer);
  }

  @Override
  public boolean onCollision(GameWorld world) {
    visible =
        game.getPlayerInventory()
            .hasItem(InventoryObject.KEYCARD); // update visibility each collision check
    if (!visible) {
      return false; // No collision if not visible
    }

    ExitConditions conditions = new ExitConditions(new String[0], new String[0], new String[0], -1);
    ExitTile exit =
        new ExitTile(
            super.getTileX(),
            super.getTileY(),
            "com.mathochist.mazegame.screen.roomscreen.LibraryBasementScreen",
            18,
            1,
            conditions);
    world.loadNewWorld(exit);
    fallSound.play();
    return true;
  }
}
