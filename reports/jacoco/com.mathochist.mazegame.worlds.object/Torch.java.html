package com.mathochist.mazegame.worlds.object;

import com.badlogic.gdx.graphics.g2d.SpriteBatch;
import com.badlogic.gdx.graphics.g2d.TextureAtlas;
import com.mathochist.mazegame.Main;
import com.mathochist.mazegame.entity.MapEntity;
import com.mathochist.mazegame.entity.Player;
import com.mathochist.mazegame.entity.playerinventory.InventoryObject;
import com.mathochist.mazegame.render.RenderBuffer;
import com.mathochist.mazegame.screen.roomscreen.RoomScreen;
import com.mathochist.mazegame.screen.roomscreen.VignetteLevel;
import com.mathochist.mazegame.userinterface.Hud;
import com.mathochist.mazegame.userinterface.speechtext.SpeechType;
import com.mathochist.mazegame.worlds.GameWorld;

/**
 * Represents a torch that when interacted with the player can see in the dark basement. This is
 * done by setting the vignette to a light one when the player interacts with the torch.
 */
public class Torch extends MapEntity {
  private boolean collected;

  /**
   * Creates the Torch object using Entity.
   *
   * @param game The instance of the game.
   * @param batch The current SpriteBatch.
   * @param atlas Which Atlas object is in.
   * @param tileX X Coord to place at.
   * @param tileY Y Coord to place at.
   * @param width Width of object.
   * @param height Height of object.
   * @param collidable Is object collidable.
   */
  public Torch(
      Main game,
      SpriteBatch batch,
      TextureAtlas atlas,
      Integer tileX,
      Integer tileY,
      Integer width,
      Integer height,
      Boolean collidable) {
    super(game, batch, atlas, "torch", tileX, tileY, width, height, collidable);

    collected = game.getPlayerInventory().hasItem(InventoryObject.TORCH);
  }

  @Override
  public boolean onInteract(Player p, GameWorld world, Hud worldHud) {
    if (collected) {
      return false;
    }
    collected = true;
    game.getPlayerInventory().addItem(InventoryObject.TORCH);
    RoomScreen screen = (RoomScreen) game.getScreen();
    screen.setVignette(VignetteLevel.light);
    worldHud.getSpeechBubbleManager().removeBubblesOfType(SpeechType.NPC_SPEECH);
    worldHud
        .getSpeechBubbleManager()
        .createSpeechBubble(SpeechType.NPC_SPEECH, "Torch Collected!", 2000);
    game.getEventsCounter().torchEvent();
    return true;
  }

  @Override
  public void render(GameWorld world, RenderBuffer renderBuffer) {
    if (collected) {
      return;
    }
    super.render(world, renderBuffer);
  }

  /**
   * Action on collison.
   *
   * @param world The game world context.
   * @return false
   */
  public boolean onCollision(GameWorld world) {
    return false;
  }

  @Override
  public void onInteractEnd(Player p, GameWorld world) {}
}
