package com.mathochist.mazegame.worlds.object;

import com.badlogic.gdx.graphics.g2d.SpriteBatch;
import com.badlogic.gdx.graphics.g2d.TextureAtlas;
import com.mathochist.mazegame.Main;
import com.mathochist.mazegame.entity.MapEntity;
import com.mathochist.mazegame.entity.Player;
import com.mathochist.mazegame.entity.playerinventory.InventoryObject;
import com.mathochist.mazegame.screen.roomscreen.RoomScreen;
import com.mathochist.mazegame.userinterface.Hud;
import com.mathochist.mazegame.userinterface.speechtext.SpeechType;
import com.mathochist.mazegame.worlds.GameWorld;

/** Creates a Vending Machine that causes the player to be slow. */
public class VendingMachine2 extends MapEntity {
  private static final long INTERACTION_COOLDOWN_MS = 2000;
  private boolean used;
  private long lastInteractionTime = 0;

  /**
   * Creates vending machine object using Entity. Also sets Vending Machine 2 as used if player
   * already has the Chamomile tea.
   *
   * @param game The instance of the game.
   * @param batch The current SpritBatch.
   * @param atlas Which Atlas object is in.
   * @param tileX X Coord to place at.
   * @param tileY Y Coord to place at.
   * @param width Width of object.
   * @param height Height of object.
   * @param collidable Is object collidable.
   */
  public VendingMachine2(
      Main game,
      SpriteBatch batch,
      TextureAtlas atlas,
      Integer tileX,
      Integer tileY,
      Integer width,
      Integer height,
      Boolean collidable) {
    super(
        game,
        batch,
        atlas,
        "vending_machine2",
        tileX,
        tileY,
        width,
        height,
        collidable); // TODO: change to Vending Machine 2

    used =
        game.getPlayerInventory()
            .hasItem(
                InventoryObject.CHAMOMILE_TEA); // If player already has Chamomile tea, mark as used
  }

  @Override
  public boolean onInteract(Player p, GameWorld world, Hud worldHud) {
    if (RoomScreen.powerOutage) {
      worldHud.getSpeechBubbleManager().removeBubblesOfType(SpeechType.NPC_SPEECH);
      worldHud
          .getSpeechBubbleManager()
          .createSpeechBubble(
              SpeechType.NPC_SPEECH, "Vending Machine doesn't work during a power outage", 2000);
      return true;
    }
    long currentTime = System.currentTimeMillis();
    if (currentTime - lastInteractionTime < INTERACTION_COOLDOWN_MS) {
      return true;
    }
    lastInteractionTime = currentTime;
    if (used) {
      worldHud.getSpeechBubbleManager().removeBubblesOfType(SpeechType.NPC_SPEECH);
      worldHud
          .getSpeechBubbleManager()
          .createSpeechBubble(SpeechType.NPC_SPEECH, "The vending machine is empty.", 2000);
      return true;
    }
    used = true;
    super.game.getPlayerInventory().addItem(InventoryObject.CHAMOMILE_TEA);
    worldHud.getSpeechBubbleManager().removeBubblesOfType(SpeechType.NPC_SPEECH);
    worldHud
        .getSpeechBubbleManager()
        .createSpeechBubble(
            SpeechType.NPC_SPEECH,
            "You got a Chamomile Tea, You are suddenly feeling tired!",
            2000);
    super.game.getEventsCounter().vendingMachineTeaEvent();
    return true;
  }

  @Override
  public void onInteractEnd(Player p, GameWorld world) {} // No action needed on interaction end

  @Override
  public boolean onCollision(GameWorld world) {
    return false; // no collision behavior
  }
}
