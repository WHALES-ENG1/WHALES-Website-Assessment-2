package com.mathochist.mazegame.worlds.object;

import com.badlogic.gdx.graphics.g2d.SpriteBatch;
import com.badlogic.gdx.graphics.g2d.TextureAtlas;
import com.mathochist.mazegame.Main;
import com.mathochist.mazegame.entity.MapEntity;
import com.mathochist.mazegame.entity.Player;
import com.mathochist.mazegame.userinterface.Hud;
import com.mathochist.mazegame.worlds.GameWorld;

/** Creates an object that appears as the book of a bookshelf to immerse the player. */
public class BookshelfTop extends MapEntity {
  /**
   * Create a bookshelf top object.
   *
   * @param game Current instance of game
   * @param batch SpriteBatch
   * @param atlas atlas containing bookshelf top
   * @param tileX Spawn X
   * @param tileY Spawn Y
   * @param width Width of bookshelf top
   * @param height height of bookshelf top
   * @param collidable is bookshelf collidable
   */
  public BookshelfTop(
      Main game,
      SpriteBatch batch,
      TextureAtlas atlas,
      Integer tileX,
      Integer tileY,
      Integer width,
      Integer height,
      Boolean collidable) {
    super(game, batch, atlas, "bookshelf_top", tileX, tileY, width, height, collidable);
  }

  @Override
  public boolean onInteract(Player p, GameWorld world, Hud worldHud) {
    return false;
  }

  @Override
  public void onInteractEnd(Player p, GameWorld world) {}

  @Override
  public boolean onCollision(GameWorld world) {
    return false;
  }
}
