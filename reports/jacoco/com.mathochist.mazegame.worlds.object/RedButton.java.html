package com.mathochist.mazegame.worlds.object;

import com.badlogic.gdx.graphics.g2d.SpriteBatch;
import com.badlogic.gdx.graphics.g2d.TextureAtlas;
import com.mathochist.mazegame.Main;
import com.mathochist.mazegame.entity.MapEntity;
import com.mathochist.mazegame.entity.Player;
import com.mathochist.mazegame.screen.roomscreen.RoomScreen;
import com.mathochist.mazegame.userinterface.Hud;
import com.mathochist.mazegame.userinterface.speechtext.SpeechType;
import com.mathochist.mazegame.worlds.GameWorld;

/**
 * An Easter egg red button that when interacted with causes a power outage.
 */
public class RedButton extends MapEntity {
  boolean interacted = false;

  /**
   * Creates Red Button object.
   *
   * @param game this instance of game
   * @param batch the spriteBatch
   * @param atlas the atlas containing the button
   * @param tileX the X coord to place at
   * @param tileY the Y coord to place at
   * @param width the width of the button
   * @param height the height of the button
   * @param collidable is the button collidable
   */
  public RedButton(
      Main game,
      SpriteBatch batch,
      TextureAtlas atlas,
      Integer tileX,
      Integer tileY,
      Integer width,
      Integer height,
      Boolean collidable) {

    super(game, batch, atlas, "red_button", tileX, tileY, width, height, collidable);

    boolean interacted = false;
  }

  @Override
  public boolean onInteract(Player p, GameWorld world, Hud worldHud) {
    if (!interacted) {
      RoomScreen.powerOutage = true;
      RoomScreen screen = (RoomScreen) game.getScreen();
      screen.getGameHud().getSpeechBubbleManager().clear();
      game.getEventsCounter().powerCutEvent();
      screen
          .getGameHud()
          .getSpeechBubbleManager()
          .createSpeechBubble(SpeechType.NPC_SPEECH, "Power went out!", 5000);
    }
    this.interacted = true;
    return false;
  }

  @Override
  public void onInteractEnd(Player p, GameWorld world) {}

  @Override
  public boolean onCollision(GameWorld world) {
    return false;
  }
}
