package com.mathochist.mazegame.entity;

import com.badlogic.gdx.Gdx;
import com.badlogic.gdx.graphics.OrthographicCamera;
import com.badlogic.gdx.graphics.g2d.Animation;
import com.badlogic.gdx.graphics.g2d.Sprite;
import com.badlogic.gdx.graphics.g2d.SpriteBatch;
import com.badlogic.gdx.graphics.g2d.TextureAtlas;
import com.badlogic.gdx.graphics.g2d.TextureRegion;
import com.mathochist.mazegame.Main;
import com.mathochist.mazegame.entity.playerinventory.InventoryObject;
import com.mathochist.mazegame.keymovement.Direction;
import com.mathochist.mazegame.keymovement.KeyBinds;
import com.mathochist.mazegame.keymovement.KeyBuffer;
import com.mathochist.mazegame.render.RenderBuffer;
import com.mathochist.mazegame.render.RenderObject;
import com.mathochist.mazegame.worlds.GameWorld;

/** Player entity controlled by user input. Handles movement, animation, and interaction. */
// TODO: Refactor to extend Entity class
public class Player {

  public static final float MOVE_SPEED = 300; // pixels per second
  public static final float SPRITE_WIDTH = 19;
  public static final float SPRITE_HEIGHT = 25;
  public static final float INTERACTION_RANGE = 2; // tiles

  /** Delay between static animation ending and control being returned to the player in seconds. */
  private static final float STATIC_ANIMATION_END_DELAY = 1.5f;

  /**
   * Delay between control being returned to player after static animation and another static
   * animation being able to be triggered.
   */
  private static final float STATIC_ANIMATION_IMMUNITY_TIME = 1.5f;

  private final Animation<TextureRegion> walkUp;
  private final Animation<TextureRegion> walkDown;
  private final Animation<TextureRegion> walkLeft;
  private final Animation<TextureRegion> walkRight;
  private final Animation<TextureRegion> fall;
  private final Sprite playerSprite;
  private final SpriteBatch screenBatch;
  private final KeyBuffer keyBuffer;
  private final OrthographicCamera camera;
  private final GameWorld world;
  private final Main game;
  private final TextureAtlas spriteSheet;
  private TextureRegion currentFrame;
  private float stateTime = 0f; // time tracker for walking animations
  private Direction direction = Direction.down;
  private float staticStateTime = 0f; // time tracker for static animations
  private float xcoord;
  private float ycoord;
  private boolean isSprinting = false;

  /** Represents whether the player is currently performing an animation that blocks movement. */
  private Animation<TextureRegion> currentStaticAnimation = null;

  /** Time left until static animations can be triggered again. */
  private float staticAnimationImmunityTimer = 0;

  /**
   * Create the player.
   *
   * @param game current game instance
   * @param camera current camera
   * @param batch SpriteBatch
   * @param world Roon
   * @param startX Spawn X coord
   * @param startY Spawn Y coord
   */
  public Player(
      Main game,
      OrthographicCamera camera,
      SpriteBatch batch,
      GameWorld world,
      float startX,
      float startY) {

    this.game = game;
    this.camera = camera;
    this.screenBatch = batch;
    this.world = world;

    // CHANGED: Using a texture atlas instead of constructing finicky and unexpandable arrays of
    // textures.
    // Load the sprite sheet as an atlas.
    spriteSheet = new TextureAtlas(Gdx.files.internal("player.atlas"));

    // Create animations for each direction
    walkUp = new Animation<>(0.15f, spriteSheet.findRegions("walk_up"));
    walkLeft = new Animation<>(0.15f, spriteSheet.findRegions("walk_left"));
    walkDown = new Animation<>(0.15f, spriteSheet.findRegions("walk_down"));
    walkRight = new Animation<>(0.15f, spriteSheet.findRegions("walk_right"));
    fall = new Animation<>(0.1f, spriteSheet.findRegions("fall"));
    // Start with first frame facing down
    currentFrame = walkDown.getKeyFrames()[0];

    // Center player on screen
    xcoord = startX;
    ycoord = startY;

    playerSprite = new Sprite(currentFrame);
    // CHANGED: Fixed player scaling issues. Player is no longer horizontally squished
    playerSprite.setScale(3.5f);
    keyBuffer = new KeyBuffer();
  }

  /**
   * Called Every frame to update player movement and interaction.
   *
   * @param gameDelta delta
   * @param renderBuffer Renderbuffer
   */
  public void update(float gameDelta, RenderBuffer renderBuffer) {
    // If a static animation is running, don't let player update.
    if (currentStaticAnimation != null) {
      updateStaticAnimation(gameDelta);
    } else {
      // Update static animation immunity timer
      staticAnimationImmunityTimer = Math.max(staticAnimationImmunityTimer - gameDelta, 0);

      keyBuffer.updateCameraMovementFromKeys();

      float moveSpeed = getMovementSpeed();

      float[] move = keyBuffer.getCameraMove();
      float deltaTime = Gdx.graphics.getDeltaTime();
      // Calculate new position and basic collision detection
      float deltaX = move[0] * deltaTime * moveSpeed;
      float deltaY = move[1] * deltaTime * moveSpeed;

      // Use player's position for collision, not camera
      boolean[] collisionLayer =
          world.getCollisionLayer(
              xcoord - currentFrame.getRegionWidth(), ycoord - currentFrame.getRegionHeight());
      if (deltaY > 0 && collisionLayer[0]) { // moving up
        deltaY = 0;
      }
      if (deltaY < 0 && collisionLayer[1]) { // moving down
        deltaY = 0;
      }
      if (deltaX < 0 && collisionLayer[2]) { // moving left
        deltaX = 0;
      }
      if (deltaX > 0 && collisionLayer[3]) { // moving right
        deltaX = 0;
      }

      if (move[0] != 0 || move[1] != 0) {
        camera.translate(deltaX, deltaY); // translate camera not player
        camera.update();
        stateTime += gameDelta;
        // Choose correct frame
        if (move[1] > 0) {
          currentFrame = walkUp.getKeyFrame(stateTime, true);
          direction = Direction.up;
        } else if (move[0] < 0) {
          currentFrame = walkLeft.getKeyFrame(stateTime, true);
          direction = Direction.left;
        } else if (move[0] > 0) {
          currentFrame = walkRight.getKeyFrame(stateTime, true);
          direction = Direction.right;
        } else if (move[1] < 0) {
          currentFrame = walkDown.getKeyFrame(stateTime, true);
          direction = Direction.down;
        }
      } else {
        stateTime = 2; // reset to idle frame
        switch (direction) {
          case up -> currentFrame = walkUp.getKeyFrame(stateTime, true);
          case down -> currentFrame = walkDown.getKeyFrame(stateTime, true);
          case left -> currentFrame = walkLeft.getKeyFrame(stateTime, true);
          case right -> currentFrame = walkRight.getKeyFrame(stateTime, true);
          default -> throw new IllegalArgumentException("Unknown Enum value.");
        }
      }
    }

    playerSprite.setRegion(currentFrame);
    playerSprite.setSize(currentFrame.getRegionWidth(), currentFrame.getRegionHeight());
    keyBuffer.clear();

    // Update player position based on camera
    this.xcoord = camera.position.x - (currentFrame.getRegionWidth() / 2f);
    this.ycoord = camera.position.y - (currentFrame.getRegionHeight() / 2f);

    this.render(renderBuffer);
  }

  /**
   * Called every frame while a static animation is running.
   *
   * @param gameDelta delta
   */
  private void updateStaticAnimation(float gameDelta) {
    staticStateTime += gameDelta;
    currentFrame = currentStaticAnimation.getKeyFrame(staticStateTime);
    if (currentStaticAnimation.isAnimationFinished(staticStateTime - STATIC_ANIMATION_END_DELAY)) {
      currentStaticAnimation = null;
      staticAnimationImmunityTimer = STATIC_ANIMATION_IMMUNITY_TIME;
    }
  }

  /**
   * Force the player into a slipping animation, preventing movement while the animation runs and
   * some time after.
   */
  public void slip() {
    if (staticAnimationImmunityTimer > 0 || currentStaticAnimation != null) {
      return;
    }
    staticStateTime = 0;
    currentStaticAnimation = fall;
    game.getEventsCounter().slipOnCoffeeEvent();
  }

  /**
   * Calculate the current movement speed of the player, based on items in inventory and sprint
   * status.
   *
   * @return A double representing the current movement speed of the player in pixels/second.
   */
  public float getMovementSpeed() {
    // base move speed buffer
    int moveSpeed = (int) MOVE_SPEED;

    // check extra keys
    if (keyBuffer.isKeyPressed(KeyBinds.SPRINT)) {
      // sprinting logic
      moveSpeed += 300;
      isSprinting = true;
      game.getAchievementHandler().playerSprinted();
    } else {
      isSprinting = false;
    }
    if (keyBuffer.isKeyPressed(KeyBinds.INTERACT)) {
      // interaction logic
      world.triggerInteractionByPixel(xcoord, ycoord, INTERACTION_RANGE, this);
    }

    // check if player has energy drink in inventory for speed boost
    if (game.getPlayerInventory().hasItem(InventoryObject.ENERGY_DRINK)) {
      moveSpeed += 150; // speed boost from energy drink
    }
    if (game.getPlayerInventory().hasItem(InventoryObject.CHAMOMILE_TEA)) {
      moveSpeed -= 150;
    }

    return moveSpeed;
  }

  /**
   * Render Player Sprite.
   *
   * @param renderBuffer to add player to
   */
  public void render(RenderBuffer renderBuffer) {
    playerSprite.setPosition(xcoord, ycoord);
    float[] tileCoords = world.pixelCoordsToTileIndex(xcoord, ycoord);
    renderBuffer.addToBuffer(
        new RenderObject(playerSprite, screenBatch, (int) Math.ceil(tileCoords[1])));
  }

  /** Dispose Used assets. */
  public void dispose() {
    spriteSheet.dispose();
  }

  public float getX() {
    return xcoord;
  }

  public void setX(float x) {
    xcoord = x;
  }

  public float getY() {
    return ycoord;
  }

  public void setY(float y) {
    ycoord = y;
  }

  public KeyBuffer getKeyBuffer() {
    return keyBuffer;
  }

  /**
   * Set position of camera.
   *
   * @param newX Center X
   * @param newY Center Y
   */
  public void setCameraPosition(float newX, float newY) {
    camera.position.set(newX, newY, 0); // Center camera
    camera.update();
  }

  public boolean isSprinting() {
    return isSprinting;
  }

  public boolean isStaticAnimationSet() {
    return currentStaticAnimation != null;
  }
}
