package com.mathochist.mazegame.screen;

import com.badlogic.gdx.Gdx;
import com.badlogic.gdx.Input;
import com.badlogic.gdx.audio.Music;
import com.badlogic.gdx.files.FileHandle;
import com.badlogic.gdx.graphics.Color;
import com.badlogic.gdx.graphics.GL32;
import com.badlogic.gdx.scenes.scene2d.Actor;
import com.badlogic.gdx.scenes.scene2d.Stage;
import com.badlogic.gdx.scenes.scene2d.ui.Button;
import com.badlogic.gdx.scenes.scene2d.ui.Label;
import com.badlogic.gdx.scenes.scene2d.ui.Table;
import com.badlogic.gdx.scenes.scene2d.utils.ChangeListener;
import com.badlogic.gdx.utils.StringBuilder;
import com.badlogic.gdx.utils.viewport.FitViewport;
import com.badlogic.gdx.utils.viewport.Viewport;
import com.mathochist.mazegame.Main;
import com.mathochist.mazegame.userinterface.ButtonHoverListener;
import com.mathochist.mazegame.userinterface.MainMenuUiSkin;
import java.util.ArrayList;

/**
 * The EndScreen class represents the End screen of the game which is used to show the player their
 * result in playing the game. It extends DefaultScreen and implements the necessary methods to
 * display a stage with UI elements.
 */
public class EndScreen extends DefaultScreen {

  public final int score = game.getScoreCounter().calculateScore();
  public final int bonusScore = game.getScoreCounter().updateBonusScore();
  private final Stage stage;
  private final boolean gameLost;
  private final String username;
  private String csvFile = "leaderboard.csv";
  private MainMenuUiSkin uiSkin;
  private Music backgroundMusic;

  /**
   * Creates the EndScreen and checks the remaining time and seconds left.
   *
   * @param game current instance of game.
   * @param backgroundMusic The currently playing background music.
   */
  public EndScreen(Main game, Music backgroundMusic) {
    super(game);
    gameLost = (game.getTimerManager().isTimeUp());
    username = game.getUsername();
    if (username == null) {
      throw new MissingUsernameException(
          "EndScreen cannot be loaded when game.username is not yet set");
    }
    game.getPlayerInventory().clear();
    Viewport viewport = new FitViewport(1280, 720);
    stage = new Stage(viewport, game.getSpriteBatch());
    this.backgroundMusic = backgroundMusic;
  }

  /**
   * Creates the EndScreen and checks the remaining time and seconds left. Also sets the CSV_FILE
   * for the leaderboard to a custom path.
   *
   * @param game current instance of game.
   * @param backgroundMusic The currently playing background music.
   * @param csvFile Path to the leaderboard CSV file.
   */
  public EndScreen(Main game, Music backgroundMusic, String csvFile) {
    this(game, backgroundMusic);
    this.csvFile = csvFile;
  }

  @Override
  public void show() {
    if (backgroundMusic == null) {
      backgroundMusic = Gdx.audio.newMusic(Gdx.files.internal("audio/title_screen.mp3"));
      backgroundMusic.setLooping(true);
      backgroundMusic.setVolume(0.5f);
      backgroundMusic.play();
    }
    Gdx.input.setInputProcessor(stage);

    uiSkin = new MainMenuUiSkin();
    Table table = new Table();
    table.setFillParent(true);
    stage.addActor(table);

    // Game over label
    Label gameOverLabel;
    if (gameLost) {
      gameOverLabel = new Label("Game Over", uiSkin.getSkin());
      table.add(gameOverLabel).padBottom(10).colspan(3);
      table.row();
      Label timeUpLabel = new Label("Time's Up! ", uiSkin.getSkin(), "presents_grey");
      table.add(timeUpLabel).padBottom(30).colspan(3);
    } else {
      gameOverLabel = new Label("You Won!", uiSkin.getSkin());
      table.add(gameOverLabel).padBottom(10).colspan(3);
      table.row();
      Label scoreLabel =
          new Label("Final score: " + (score + bonusScore), uiSkin.getSkin(), "game_font");
      scoreLabel.setFontScale(1.2f);
      table.add(scoreLabel).padBottom(30).colspan(3);
    }

    // Leaderboard
    table.row();
    table.add(new Label("RANK", uiSkin.getSkin(), "game_font")).padRight(20);
    table.add(new Label("NAME", uiSkin.getSkin(), "game_font")).padRight(50);
    table.add(new Label("SCORE", uiSkin.getSkin(), "game_font")).row();
    table
        .add(new Label("- - - - - - - - - - - - - - - - - -", uiSkin.getSkin(), "game_font"))
        .colspan(3)
        .padBottom(10)
        .row();

    int rank = 1;
    ArrayList<String[]> leaderBoard =
        updateLeaderboard(gameLost ? null : username, score + bonusScore);
    for (String[] row : leaderBoard) {
      String name = row[0];
      String scoreStr = row[1];
      Label rankLabel = new Label(rank + ".", uiSkin.getSkin(), "game_font");
      Label nameLabel = new Label(name, uiSkin.getSkin(), "game_font");
      Label scoreLabel = new Label(scoreStr, uiSkin.getSkin(), "game_font");
      boolean isPlayer =
          !gameLost && name.equals(username) && scoreStr.equals(String.valueOf(score + bonusScore));
      if (isPlayer) {
        rankLabel.setColor(Color.GOLD);
        nameLabel.setColor(Color.GOLD);
        scoreLabel.setColor(Color.GOLD);
      }
      table.add(rankLabel).padRight(20);
      table.add(nameLabel).padRight(50);
      table.add(scoreLabel).row();
      rank++;
    }
    Button backToMenuButton = new Button(uiSkin.getSkin(), "back_to_MM");
    backToMenuButton.addListener(
        new ChangeListener() {
          @Override
          public void changed(ChangeListener.ChangeEvent changeEvent, Actor actor) {
            backgroundMusic.stop();
            game.setScreen(new TitleScreen(game));
          }
        });
    backToMenuButton.row();

    backToMenuButton.addListener(new ButtonHoverListener());
    table.add(backToMenuButton).width(250).height(50).padTop(30).colspan(3);
  }

  /**
   * Update the leaderboard file with the given username and score. Will not change if the score
   * isn't high enough to reach the leaderboard.
   *
   * @param playerName The username of the player to use for the score.
   * @param totalScore The score achieved by the player.
   * @return The new leaderboard as an array list containing 10 elements, each one is a 2-array
   *     <code>[username, score]</code>
   */
  public ArrayList<String[]> updateLeaderboard(String playerName, int totalScore) {
    FileHandle file = Gdx.files.local(csvFile);
    ArrayList<String[]> scores = new ArrayList<>();
    if (file.exists()) {
      String[] lines = file.readString().split("\\r?\\n");
      for (String line : lines) {
        String[] parts = line.split(",");
        if (parts.length == 2) {
          scores.add(parts);
        }
      }
    }
    if (playerName != null && !playerName.isEmpty()) {
      scores.add(new String[] {playerName, String.valueOf(totalScore)});
    }
    scores.sort(
        (o1, o2) -> {
          int s1 = 0;
          int s2 = 0;
          try {
            s1 = Integer.parseInt(o1[1]);
          } catch (NumberFormatException e) {
            System.err.println(e.getMessage());
          }
          try {
            s2 = Integer.parseInt(o2[1]);
          } catch (NumberFormatException e) {
            System.err.println(e.getMessage());
          }
          return Integer.compare(s2, s1);
        });

    StringBuilder sb = new StringBuilder();
    ArrayList<String[]> top10 = new ArrayList<>();
    int count = 0;
    for (String[] row : scores) {
      if (count >= 10) {
        break;
      }
      top10.add(row);
      sb.append(row[0]).append(",").append(row[1]).append("\n");
      count++;
    }
    if (playerName != null) {
      file.writeString(sb.toString(), false);
    }

    return top10;
  }

  @Override
  public void render(float delta) {
    Gdx.gl.glClearColor(0, 0, 0, 1);
    Gdx.gl.glClear(GL32.GL_COLOR_BUFFER_BIT);
    // AchievementScreen is accessible from EndScreen
    if (Gdx.input.isKeyJustPressed(Input.Keys.TAB)) {
      game.setScreen(new AchievementScreen(game, this));
      return;
    }
    stage.act(delta);
    stage.draw();
  }

  @Override
  public void renderShapes() {}

  @Override
  public void resize(int width, int height) {
    stage.getViewport().update(width, height, true);
  }

  @Override
  public void dispose() {
    if (stage != null) {
      stage.dispose();
    }
    if (uiSkin != null) {
      uiSkin.dispose();
    }
    if (backgroundMusic != null) {
      backgroundMusic.dispose();
    }
  }

  @Override
  public boolean keyDown(int keycode) {
    return false;
  }

  @Override
  public boolean keyUp(int keycode) {
    return false;
  }
}
