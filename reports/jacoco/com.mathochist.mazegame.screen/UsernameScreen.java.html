package com.mathochist.mazegame.screen;

import com.badlogic.gdx.Gdx;
import com.badlogic.gdx.Input;
import com.badlogic.gdx.audio.Music;
import com.badlogic.gdx.files.FileHandle;
import com.badlogic.gdx.graphics.Cursor;
import com.badlogic.gdx.graphics.GL32;
import com.badlogic.gdx.scenes.scene2d.InputEvent;
import com.badlogic.gdx.scenes.scene2d.InputListener;
import com.badlogic.gdx.scenes.scene2d.Stage;
import com.badlogic.gdx.scenes.scene2d.ui.Label;
import com.badlogic.gdx.scenes.scene2d.ui.Table;
import com.badlogic.gdx.scenes.scene2d.utils.ClickListener;
import com.badlogic.gdx.utils.viewport.FitViewport;
import com.mathochist.mazegame.Main;
import com.mathochist.mazegame.userinterface.MainMenuUiSkin;

/** Screen to allow the user to enter a username to record a new high score. */
public class UsernameScreen extends DefaultScreen {

  private final Music backgroundMusic;
  private final String[] profanity;
  private Stage stage;
  private MainMenuUiSkin uiSkin;
  private String enteredName = "";

  private Label errorLabel;

  /**
   * Create a new username screen.
   *
   * @param main Current instance of Main.
   */
  public UsernameScreen(Main main) {
    super(main);

    Gdx.graphics.setSystemCursor(Cursor.SystemCursor.Arrow);
    backgroundMusic = Gdx.audio.newMusic(Gdx.files.internal("audio/title_screen.mp3"));

    FileHandle file = Gdx.files.internal("game_ui/profanity.txt");
    profanity = file.readString().split("\\r?\\n");
  }

  @Override
  public boolean keyDown(int keycode) {
    return false;
  }

  @Override
  public boolean keyUp(int keycode) {
    return false;
  }

  @Override
  public void renderShapes() {}

  @Override
  public void show() {
    // Skip the screen if a username is already set for this session.
    if (game.getUsername() != null) {
      nextScreen();
    }

    stage = new Stage(new FitViewport(1280, 720), game.getSpriteBatch());
    Gdx.input.setInputProcessor(stage);
    uiSkin = new MainMenuUiSkin();
    showNameInputUserInterface();
  }

  @Override
  public void render(float delta) {
    Gdx.gl.glClearColor(0, 0, 0, 1);
    Gdx.gl.glClear(GL32.GL_COLOR_BUFFER_BIT);
    stage.act(delta);
    stage.draw();

    if (Gdx.input.isKeyJustPressed(Input.Keys.ENTER)) {
      handleSubmit();
    }
  }

  @Override
  public void resize(int width, int height) {
    stage.getViewport().update(width, height, true);
  }

  private void showNameInputUserInterface() {
    stage.clear();
    enteredName = "";
    Table nameTable = new Table();
    nameTable.setFillParent(true);
    stage.addActor(nameTable);

    // Create label asking user to enter their name
    Label enterName = new Label("Enter name:", uiSkin.getSkin());
    enterName.setFontScale(0.5f);
    nameTable.add(enterName).row();

    // Create label showing the user's entered name
    Label nameField = new Label("____", uiSkin.getSkin());
    nameField.setFontScale(0.5f);
    nameTable.add(nameField).pad(20).row();

    // Create label showing any errors from the user's entered name
    errorLabel = new Label("", uiSkin.getSkin());
    errorLabel.setColor(1, 0, 0, 1);
    errorLabel.setFontScale(0.3f);
    nameTable.add(errorLabel).padBottom(10).row();

    // Create clickable label to submit the name
    Label submitButton = new Label("Submit", uiSkin.getSkin());
    submitButton.setFontScale(0.3f);
    nameTable.add(submitButton).padBottom(20).row();

    // Add listener to handle user typing
    stage.addListener(
        new InputListener() {
          @Override
          public boolean keyTyped(InputEvent event, char character) {
            if (character == '\b' && !enteredName.isEmpty()) {
              enteredName = enteredName.substring(0, enteredName.length() - 1);
            } else if (enteredName.length() < 4 && Character.isLetter(character)) {
              enteredName += Character.toUpperCase(character);
            }
            nameField.setText(enteredName + "_".repeat(4 - enteredName.length()));
            return true;
          }
        });

    // Add listener to handle user clicking submit
    submitButton.addListener(
        new ClickListener() {
          @Override
          public void clicked(InputEvent event, float x, float y) {
            handleSubmit();
          }
        });
  }

  private void handleSubmit() {
    if (enteredName.length() != 4) {
      errorLabel.setText("Name must be 4 characters long!");
      return;
    }
    for (String badWord : profanity) {
      badWord = badWord.trim().toUpperCase();
      if (enteredName.contains(badWord)) {
        errorLabel.setText("Invalid Name!");
        return;
      }
    }
    game.setUsername(enteredName);
    nextScreen();
  }

  private void nextScreen() {
    backgroundMusic.dispose();
    game.setScreen(new EndScreen(game, backgroundMusic));
  }

  @Override
  public void dispose() {
    stage.dispose();
    uiSkin.dispose();
  }
}
