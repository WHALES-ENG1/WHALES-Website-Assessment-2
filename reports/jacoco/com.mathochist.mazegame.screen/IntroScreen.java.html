package com.mathochist.mazegame.screen;

import com.badlogic.gdx.Gdx;
import com.badlogic.gdx.Input;
import com.badlogic.gdx.audio.Music;
import com.badlogic.gdx.files.FileHandle;
import com.badlogic.gdx.graphics.Cursor;
import com.badlogic.gdx.graphics.GL32;
import com.badlogic.gdx.graphics.Texture;
import com.badlogic.gdx.scenes.scene2d.InputEvent;
import com.badlogic.gdx.scenes.scene2d.InputListener;
import com.badlogic.gdx.scenes.scene2d.Stage;
import com.badlogic.gdx.scenes.scene2d.ui.Image;
import com.badlogic.gdx.scenes.scene2d.ui.Label;
import com.badlogic.gdx.scenes.scene2d.ui.Table;
import com.badlogic.gdx.scenes.scene2d.utils.ClickListener;
import com.badlogic.gdx.utils.viewport.FitViewport;
import com.mathochist.mazegame.Main;
import com.mathochist.mazegame.screen.roomscreen.MorrellScreen;
import com.mathochist.mazegame.userinterface.MainMenuUiSkin;

/**
 * The IntroScreen class represents the Intro screen of the game which is used to display the
 * instructions for how the user can interact with the game. It extends DefaultScreen and implements
 * the necessary methods to display a stage with UI elements.
 */
public class IntroScreen extends DefaultScreen {

  private final Music backgroundMusic;
  private Stage stage;
  private MainMenuUiSkin uiSkin;

  /**
   * Creates the Intro Screen.
   *
   * @param game Current instance of game
   * @param backgroundMusic Background Music to play
   */
  public IntroScreen(Main game, Music backgroundMusic) {
    super(game);
    Gdx.graphics.setSystemCursor(Cursor.SystemCursor.Arrow);
    this.backgroundMusic = backgroundMusic;
  }

  @Override
  public void show() {
    stage = new Stage(new FitViewport(1280, 720), game.getSpriteBatch());
    Gdx.input.setInputProcessor(stage);
    uiSkin = new MainMenuUiSkin();
    showStoryUserInterface();
  }

  private void showStoryUserInterface() {
    stage.clear();
    Table table = new Table();
    table.setFillParent(true);
    stage.addActor(table);
    Label settingSceneLabel =
        new Label(
            "You are studying in the Morrell Library after closing hours, not during "
                + "exam season, and the dean sees you!",
            uiSkin.getSkin(),
            "game_font");
    settingSceneLabel.setWrap(true);
    settingSceneLabel.setAlignment(0, 0); // Center the writing
    Label instructionLabel =
        new Label(
            "Find your lost keycard and get out before you get caught!",
            uiSkin.getSkin(),
            "game_font");
    table.add(settingSceneLabel).width(800).padTop(100).padBottom(50);
    table.row();
    table.add(instructionLabel).padBottom(10);
    table.row();
    Label timeLabel =
        new Label("You have 5 minutes to escape the uni!", uiSkin.getSkin(), "game_font");
    table.add(timeLabel).padBottom(70);
    table.row();
    Texture keyBindsTexture = new Texture(Gdx.files.internal("MainMenu/KeyBinds.png"));
    Image keyBindsImage = new Image(keyBindsTexture);
    table.add(keyBindsImage).size(600, 200).padBottom(50);
    table.row();
    Label skipLabel = new Label("Press Any Button to continue..", uiSkin.getSkin(), "game_font");
    table.add(skipLabel).expand().right().padRight(50).padBottom(50);
  }

  @Override
  public void render(float delta) {
    Gdx.gl.glClearColor(0, 0, 0, 1);
    Gdx.gl.glClear(GL32.GL_COLOR_BUFFER_BIT);
    stage.act(delta);
    stage.draw();
    if (Gdx.input.isKeyJustPressed(Input.Keys.ANY_KEY)) {
      backgroundMusic.dispose();
      game.getTimerManager().startTimer();
      game.setScreen(new MorrellScreen(game));
    }
  }

  @Override
  public void renderShapes() {}

  @Override
  public void resize(int width, int height) {
    stage.getViewport().update(width, height, true);
  }

  @Override
  public void dispose() {
    stage.dispose();
    uiSkin.dispose();
  }

  @Override
  public boolean keyDown(int keycode) {
    return false;
  }

  @Override
  public boolean keyUp(int keycode) {
    return false;
  }
}
