package com.mathochist.mazegame.screen;

import com.badlogic.gdx.Gdx;
import com.badlogic.gdx.Input;
import com.badlogic.gdx.Screen;
import com.badlogic.gdx.graphics.Color;
import com.badlogic.gdx.graphics.GL32;
import com.badlogic.gdx.graphics.g2d.TextureAtlas;
import com.badlogic.gdx.graphics.g2d.TextureRegion;
import com.badlogic.gdx.scenes.scene2d.Stage;
import com.badlogic.gdx.scenes.scene2d.ui.Image;
import com.badlogic.gdx.scenes.scene2d.ui.Label;
import com.badlogic.gdx.scenes.scene2d.ui.Table;
import com.badlogic.gdx.utils.Align;
import com.badlogic.gdx.utils.viewport.FitViewport;
import com.mathochist.mazegame.Main;
import com.mathochist.mazegame.userinterface.MainMenuUiSkin;

/**
 * Achievement Screen used to display Achievement progression to the user. Displays during the game
 * when the user hits TAB. Can return back from this Screen into the game by hitting TAB again.
 */
public class AchievementScreen extends DefaultScreen {

  private final Stage stage;
  private final Screen parentScreen;
  private final TextureAtlas achievementAtlas;
  private MainMenuUiSkin uiSkin;

  /**
   * Creates the Achievement Screen to display.
   *
   * @param game Current instance of game
   * @param parentScreen The Screen the game should return to
   */
  public AchievementScreen(Main game, Screen parentScreen) {
    super(game);
    this.parentScreen = parentScreen;
    stage = new Stage(new FitViewport(1280, 720), game.getSpriteBatch());
    achievementAtlas =
        new TextureAtlas(Gdx.files.internal("game_ui/achievements/achievements.atlas"));
  }

  @Override
  public void show() {
    Gdx.input.setInputProcessor(stage);
    uiSkin = new MainMenuUiSkin();

    Table table = new Table();
    table.setFillParent(true);
    stage.addActor(table);
    // table.setDebug(true);

    // Add UI elements to the table using uiSkin
    Label titleLabel = new Label("Achievements", uiSkin.getSkin(), "game_font");
    titleLabel.setFontScale(1.5f);
    table.add(titleLabel).colspan(5).padBottom(50).center();
    table.row();
    // Displays square icons to track progress
    boolean allEventsDone = game.getAchievementHandler().isAllEventsTriggeredEarned();
    table
        .add(
            createAchievementSquare(
                "All Events ", "all_events_triggered", allEventsDone, "Trigger all events"))
        .pad(20);
    boolean fastEscapeDone = game.getAchievementHandler().isFastEscapeEarned();
    table
        .add(
            createAchievementSquare(
                "Fast Escape", "fast_escape", fastEscapeDone, "Escape within 2:30"))
        .pad(20);
    boolean noNegativesDone = game.getAchievementHandler().isNoNegativeEventsEarned();
    table
        .add(
            createAchievementSquare(
                "Clean Sheet", "no_negatives", noNegativesDone, "No negative Events"))
        .pad(20);
    boolean slowAndSteadyDone = game.getAchievementHandler().isSlowAndSteadyEarned();
    table
        .add(
            createAchievementSquare(
                "Slow and Steady", "slow_and_steady", slowAndSteadyDone, "Don't sprint"))
        .pad(20);
    boolean longBoiDone = game.getAchievementHandler().isDuckFound();
    table
        .add(createAchievementSquare("Long Boi", "long_boi", longBoiDone, "Find Long Boi"))
        .pad(20);

    table.row();

    Label backLabel = new Label("Press TAB to return", uiSkin.getSkin(), "game_font");
    backLabel.setColor(Color.GRAY);
    table.add(backLabel).colspan(5).padTop(100);
  }

  /**
   * Helper method to clean up repeated Code.
   *
   * @param title Used to Label the Achievement.
   * @param regionName What the square is called in the atlas.
   * @param isUnlocked Whether Achievement has been completed.
   * @param description How to trigger Achievements
   */
  private Table createAchievementSquare(
      String title, String regionName, boolean isUnlocked, String description) {

    TextureRegion region = achievementAtlas.findRegion(regionName);
    Image icon = new Image(region);

    // Greyed out if Achievement isn't complete
    if (!isUnlocked) {
      icon.setColor(new Color(0.2f, 0.2f, 0.2f, 1f));
    }
    Table square = new Table();
    Label nameLabel = new Label(title, uiSkin.getSkin(), "game_font");
    Label descriptionLabel = new Label(description, uiSkin.getSkin(), "game_font");
    descriptionLabel.setFontScale(0.5f);
    nameLabel.setAlignment(Align.center);
    square.add(nameLabel).row();
    square.add(icon).size(128, 128).padBottom(10).row();
    square.add(descriptionLabel).row();

    return square;
  }

  @Override
  public boolean keyDown(int i) {
    return false;
  }

  @Override
  public boolean keyUp(int i) {
    return false;
  }

  @Override
  public void renderShapes() {}

  @Override
  public void render(float v) {
    Gdx.gl.glClearColor(0, 0, 0, 1);
    Gdx.gl.glClear(GL32.GL_COLOR_BUFFER_BIT);

    if (Gdx.input.isKeyJustPressed(Input.Keys.TAB)) {
      if (parentScreen != null) {
        game.setScreen(parentScreen);
      } else {
        game.setScreen(new TitleScreen(game));
      }
      return;
    }
    stage.act(v);
    stage.draw();
  }

  @Override
  public void resize(int width, int height) {
    stage.getViewport().update(width, height, true);
  }

  @Override
  public void dispose() {
    stage.dispose();
    uiSkin.dispose();

    if (achievementAtlas != null) {
      achievementAtlas.dispose();
    }
  }
}
