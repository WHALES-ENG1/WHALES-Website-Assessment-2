package com.mathochist.mazegame.userinterface.speechtext;

import com.badlogic.gdx.graphics.g2d.BitmapFont;
import com.mathochist.mazegame.userinterface.Hud;
import java.util.ArrayList;

/** Manages the Speech Bubbles. */
public class SpeechBubbleManager {
  private final Hud hud;
  private final ArrayList<SpeechBubble> bubbles;

  /**
   * Creates the Speech Bubble Manager. Create bubbles array.
   *
   * @param hud Current instance of HUD
   */
  public SpeechBubbleManager(Hud hud) {
    this.hud = hud;
    this.bubbles = new ArrayList<>();
  }

  /** Display Bubbles to Screen. */
  public void render() {
    ArrayList<SpeechBubble> bubblesToRemove = new ArrayList<>();
    for (SpeechBubble bubble : bubbles) {
      bubble.render();
      if (bubble.isExpired()) {
        bubble.dispose();
        bubblesToRemove.add(bubble);
      }
    }
    bubbles.removeAll(bubblesToRemove);
  }

  /**
   * Create Speech bubble to display.
   *
   * @param type What Speech type is it
   * @param message Message to display
   * @param durationMs How long message should list
   */
  public void createSpeechBubble(SpeechType type, String message, int durationMs) {
    SpeechBubble bubble = new SpeechBubble(type, message, durationMs, hud.getMainFont(), hud);
    bubble.setStartTime(System.currentTimeMillis());
    bubbles.add(bubble);
  }

  /**
   * Create Speech bubble to display with a different font.
   *
   * @param type What Speech type is it
   * @param message Message to display
   * @param durationMs How long message should list
   * @param font new font
   */
  public void createSpeechBubble(SpeechType type, String message, int durationMs, BitmapFont font) {
    SpeechBubble bubble = new SpeechBubble(type, message, durationMs, font, hud);
    bubble.setStartTime(System.currentTimeMillis());
    bubbles.add(bubble);
  }

  /** dispose used assets. */
  public void dispose() {
    for (SpeechBubble bubble : bubbles) {
      bubble.dispose();
    }
    bubbles.clear();
  }

  /**
   * Remove bubbles.
   *
   * @param type what type of bubbles to remove
   */
  public void removeBubblesOfType(SpeechType type) {
    ArrayList<SpeechBubble> bubblesToRemove = new ArrayList<>();
    for (SpeechBubble bubble : bubbles) {
      if (bubble.getType() == type) {
        bubble.dispose();
        bubblesToRemove.add(bubble);
      }
    }
    bubbles.removeAll(bubblesToRemove);
  }

  /** dispose all bubbles. */
  public void clear() {
    for (SpeechBubble bubble : bubbles) {
      bubble.dispose();
    }
    bubbles.clear();
  }
}
