package com.mathochist.mazegame.userinterface.speechtext;

import com.badlogic.gdx.graphics.Color;
import com.badlogic.gdx.graphics.Pixmap;
import com.badlogic.gdx.graphics.Texture;
import com.badlogic.gdx.graphics.g2d.BitmapFont;
import com.badlogic.gdx.graphics.g2d.GlyphLayout;
import com.badlogic.gdx.graphics.g2d.TextureAtlas;
import com.mathochist.mazegame.userinterface.Hud;
import java.util.Objects;

/**
 * SpeechBubble used to guide or warn the player in the game and to confirm objects they have
 * interacted with.
 */
public class SpeechBubble {

  private final Hud hud;
  private final SpeechType type;
  private final String message;
  private final int durationMs;
  private final BitmapFont font;
  private final Texture pixel;
  private double startTime;

  // CHANGED: Uses sprite batch for all speech bubble rendering, instead of shape renderer.
  // This was done to avoid issues with the order of centralised rendering covering text from speech
  // bubbles

  /**
   * Create a SpeechBubble.
   *
   * @param type What SpeechType
   * @param message the message
   * @param durationMs the length to display it for
   * @param font the font
   * @param hud current instance of HUD.
   */
  public SpeechBubble(SpeechType type, String message, int durationMs, BitmapFont font, Hud hud) {
    this.type = type;
    this.message = message;
    this.durationMs = durationMs;
    this.font = font;
    this.hud = hud;

    // Create a single pixel texture to use to draw rectangles
    Pixmap pm = new Pixmap(1, 1, Pixmap.Format.RGBA8888);
    pm.setColor(Color.WHITE);
    pm.fill();
    this.pixel = new Texture(pm);
    pm.dispose();
  }

  private void renderNpcBubble() {
    GlyphLayout layout = new GlyphLayout(font, message);

    float padding = 40.0f;
    float bubbleWidth = layout.width + padding * 2;
    float bubbleHeight = layout.height + padding * 2;
    float bubbleX = (hud.getHudCamera().viewportWidth - bubbleWidth) / 2.0f;
    float bubbleY = hud.getHudCamera().viewportHeight - bubbleHeight - 40.0f;

    // Draw rectangle
    hud.getHudBatch().setColor(Color.BLACK); // Black rectangle
    hud.getHudBatch().draw(pixel, bubbleX, bubbleY, bubbleWidth, bubbleHeight);
    hud.getHudBatch().setColor(Color.WHITE); // Reset

    // Draw carriage return arrow below speech bubble.
    TextureAtlas.AtlasRegion carriageReturnAsset =
        hud.getGameUiAtlas().findRegion("speech_bubble_carriage_return");
    float arrowWidth = carriageReturnAsset.originalWidth / 1.5f;
    float arrowHeight = carriageReturnAsset.originalHeight / 1.5f;
    float arrowX = hud.getHudCamera().viewportWidth / 2.0f - arrowWidth / 2.0f;
    float arrowY =
        (float) (bubbleY + 5.0f * Math.sin(System.currentTimeMillis() / 200.0))
            - arrowHeight / 2.0f;
    hud.getHudBatch().draw(carriageReturnAsset, arrowX, arrowY, arrowWidth, arrowHeight);

    // Draw text
    font.draw(hud.getHudBatch(), message, bubbleX + padding, bubbleY + bubbleHeight - padding);
  }

  private void pass() {}

  // TODO: Add more speech bubble types and their render methods

  /** render NPC Speech. */
  public void render() {
    if (Objects.requireNonNull(type) == SpeechType.NPC_SPEECH) {
      renderNpcBubble();
    } else {
      pass();
    }
  }

  public SpeechType getType() {
    return type;
  }

  public String getMessage() {
    return message;
  }

  public int getDurationMs() {
    return durationMs;
  }

  public BitmapFont getFont() {
    return font;
  }

  public void setStartTime(double startTime) {
    this.startTime = startTime;
  }

  public boolean isExpired() {
    return (System.currentTimeMillis() - startTime) >= durationMs;
  }

  /** Dispose Used Assets. */
  public void dispose() {}
}
