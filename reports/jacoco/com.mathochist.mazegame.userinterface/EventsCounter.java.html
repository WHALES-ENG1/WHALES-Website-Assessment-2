package com.mathochist.mazegame.userinterface;

import com.badlogic.gdx.graphics.OrthographicCamera;
import com.badlogic.gdx.graphics.g2d.BitmapFont;
import com.badlogic.gdx.graphics.g2d.SpriteBatch;
import com.mathochist.mazegame.Main;

/** Keeps track of the Events and progression with them. */
public class EventsCounter {
  private final Main game;
  private int positiveEvents;
  private int negativeEvents;
  private int hiddenEvents;
  private boolean deanCatchEventTriggered;
  private boolean librarySprintEventTriggered;
  private boolean keycardNeededEventTriggered;
  private boolean vendingMachineEnergyEventTriggered;
  private boolean vendingMachineTeaEventTriggered;
  private boolean fallenIntoBasementEventTriggered;
  private boolean torchEventTriggered;
  private boolean glassesEventTriggered;
  private boolean slipOnCoffeeEventTriggered;
  private boolean fireAlarmEventTriggered;
  private boolean powerCutEventTriggered;

  /**
   * Fresh setup of Events (creates Event counter).
   *
   * @param game current instance of game
   */
  public EventsCounter(Main game) {
    this.positiveEvents = 0;
    this.negativeEvents = 0;
    this.hiddenEvents = 0;
    this.game = game;
    this.librarySprintEventTriggered = false;
    this.fireAlarmEventTriggered = false;
  }

  /** Call when library sprint event has happened to update hidden counter. */
  public void librarySprintEvent() {
    if (librarySprintEventTriggered) {
      return;
    }
    this.hiddenEvents += 1;
    this.librarySprintEventTriggered = true;
  }

  /** Call when Keycard needed event has happened to update negative counter. */
  public void keycardNeededEvent() {
    if (keycardNeededEventTriggered) {
      return;
    }
    this.negativeEvents += 1;
    this.keycardNeededEventTriggered = true;
    game.getAchievementHandler().triggerNegative();
    game.getScoreCounter().changeScore(-5f);
  }

  /** Call when Vending Machine event has happened to update positive counter. */
  public void vendingMachineEnergyEvent() {
    if (vendingMachineEnergyEventTriggered) {
      return;
    }
    this.positiveEvents += 1;
    this.vendingMachineEnergyEventTriggered = true;
    game.getScoreCounter().changeScore(10f);
  }

  /** Call when Vending Machine event has happened to update positive counter. */
  public void vendingMachineTeaEvent() {
    if (vendingMachineTeaEventTriggered) {
      return;
    }
    this.negativeEvents += 1;
    this.vendingMachineTeaEventTriggered = true;
    game.getScoreCounter().changeScore(-10f);
  }

  /** Call when torch event has happened to update positive counter. */
  public void torchEvent() {
    if (torchEventTriggered) {
      return;
    }
    this.positiveEvents += 1;
    this.torchEventTriggered = true;
    game.getScoreCounter().changeScore(15f);
  }

  /** Call when glasses event has happened to update positive counter. */
  public void glassesEvent() {
    if (glassesEventTriggered) {
      return;
    }
    this.positiveEvents += 1;
    this.glassesEventTriggered = true;
    game.getScoreCounter().changeScore(10f);
  }

  /** Call when fallen into basement event has happened to update hidden counter. */
  public void fallenIntoBasementEvent() {
    if (fallenIntoBasementEventTriggered) {
      return;
    }
    this.hiddenEvents += 1;
    this.fallenIntoBasementEventTriggered = true;
  }

  /** Call when FireAlarmEvent is active. */
  public void fireEvent() {
    if (fireAlarmEventTriggered) {
      return;
    }
    this.hiddenEvents += 1;
    this.fireAlarmEventTriggered = true;
  }

  /** Call when Dean Catches to update negative event counter. */
  public void deanCatchEvent() {
    if (deanCatchEventTriggered) {
      return;
    }
    this.negativeEvents += 1;
    this.deanCatchEventTriggered = true;
    game.getScoreCounter().changeScore(-25f);
  }

  /** Call when slipped on coffee to update negative event counter. */
  public void slipOnCoffeeEvent() {
    if (slipOnCoffeeEventTriggered) {
      return;
    }
    this.negativeEvents += 1;
    this.slipOnCoffeeEventTriggered = true;
    game.getScoreCounter().changeScore(-10f);
  }

  /** Call when power cut event has been triggered. */
  public void powerCutEvent() {
    if (powerCutEventTriggered) {
      return;
    }
    this.negativeEvents += 1;
    this.powerCutEventTriggered = true;
    game.getScoreCounter().changeScore(-50f);
  }

  /**
   * Renders Counter using HUD batch.
   *
   * @param hudBatch HUD Batch
   * @param camera current Camera
   * @param font current font
   */
  public void renderEventsCounterToHud(
      SpriteBatch hudBatch, OrthographicCamera camera, BitmapFont font) {
    font.draw(hudBatch, "Positive Events: " + positiveEvents, camera.viewportWidth - 125, 60);
    font.draw(hudBatch, "Negative Events: " + negativeEvents, camera.viewportWidth - 125, 45);
    font.draw(hudBatch, "Hidden Events: " + hiddenEvents, camera.viewportWidth - 125, 30);
  }

  public int getPositiveEvents() {
    return positiveEvents;
  }

  public int getNegativeEvents() {
    return negativeEvents;
  }

  public int getHiddenEvents() {
    return hiddenEvents;
  }

  public boolean isLibrarySprintEventTriggered() {
    return librarySprintEventTriggered;
  }

  public boolean isKeycardNeededEventTriggered() {
    return keycardNeededEventTriggered;
  }

  public boolean isVendingMachineEventTriggered() {
    return vendingMachineEnergyEventTriggered;
  }

  public boolean isFallenIntoBasementEventTriggered() {
    return fallenIntoBasementEventTriggered;
  }

  public boolean isPowerCutEventTriggered() {
    return powerCutEventTriggered;
  }

  public boolean isFireAlarmEventTriggered() {
    return fireAlarmEventTriggered;
  }

  public boolean isVendingMachineTeaEventTriggered() {
    return vendingMachineTeaEventTriggered;
  }

  public boolean isDeanCatchEventTriggered() {
    return deanCatchEventTriggered;
  }

  public boolean isGlassesEventTriggered() {
    return glassesEventTriggered;
  }

  public boolean isTorchEventTriggered() {
    return torchEventTriggered;
  }

  public boolean isSlipOnCoffeeEventTriggered() {
    return slipOnCoffeeEventTriggered;
  }

  public boolean isAllEventsTriggered() {
    return (this.positiveEvents + this.negativeEvents + this.hiddenEvents) == 11;
  }
}
