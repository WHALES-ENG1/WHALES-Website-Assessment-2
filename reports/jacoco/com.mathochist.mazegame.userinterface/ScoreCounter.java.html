package com.mathochist.mazegame.userinterface;

import com.badlogic.gdx.graphics.OrthographicCamera;
import com.badlogic.gdx.graphics.g2d.BitmapFont;
import com.badlogic.gdx.graphics.g2d.SpriteBatch;
import com.mathochist.mazegame.Main;

/** A class that keeps track of the score to be displayed in the hud and at the end Screen. */
public class ScoreCounter {
  private final Main game;
  private int bonusScore = 0;
  private float score;
  private float lastChangeAmount = 0;
  private long lastChangeTime = 0;

  /** Creates a Score Counter and sets the initial score to 300 (seconds the game starts at). */
  public ScoreCounter(Main game) {
    this.game = game;
  }

  /**
   * Calculate the current score.
   *
   * @return (event scores + time remaining) as an integer.
   */
  public int calculateScore() {
    return (int) (score + game.getTimerManager().getTimeLeft());
  }

  public float getScore() {
    return score;
  }

  public int getBonusScore() {
    return bonusScore;
  }

  /**
   * render score to hud.
   *
   * @param font hud font
   * @param batch hud spriteBatch
   * @param camera game camera
   */
  public void renderToHud(BitmapFont font, SpriteBatch batch, OrthographicCamera camera) {
    font.draw(
        batch,
        "Score: " + calculateScore(),
        camera.viewportWidth / 2 - 50,
        camera.viewportHeight - 2);

    if (System.currentTimeMillis() - lastChangeTime < 2000) {
      if (lastChangeAmount >= 0) {
        font.setColor(0, 1, 0, 1);
      } else {
        font.setColor(1, 0, 0, 1);
      }

      String text = (lastChangeAmount >= 0 ? "+" : "") + (int) lastChangeAmount;
      font.draw(batch, text, camera.viewportWidth / 2 + 80, camera.viewportHeight - 2);
      font.setColor(1, 1, 1, 1);
    }
  }

  /**
   * Change Score by change amount.
   *
   * @param change Change amount, negative to decrement, positive to increase
   */
  public void changeScore(float change) {
    this.score += change;

    this.lastChangeAmount = change;
    this.lastChangeTime = System.currentTimeMillis();
  }

  /** add score from completing achievements. */
  public int updateBonusScore() {
    bonusScore = 0;
    if (game.getAchievementHandler().isAllEventsTriggeredEarned()) {
      bonusScore += 450;
    }
    if (game.getAchievementHandler().isFastEscapeEarned()) {
      bonusScore += 100;
    }
    if (game.getAchievementHandler().isNoNegativeEventsEarned()) {
      bonusScore += 75;
    }
    if (game.getAchievementHandler().isSlowAndSteadyEarned()) {
      bonusScore += 125;
    }
    if (game.getAchievementHandler().isDuckFound()) {
      bonusScore += 100;
    }
    return bonusScore;
  }
}
