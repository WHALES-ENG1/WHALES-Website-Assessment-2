package com.mathochist.mazegame.userinterface;

import com.mathochist.mazegame.Main;

/**
 * This class tracks Achievements. It is used primarily by the Achievement Screen to receive
 * up-to-date progress.
 */
public class AchievementHandler {
  private final Main game;
  private boolean sprintUsed;
  private boolean noNegativeEvents;
  private boolean duckFound;

  /**
   * Creates Achievement Handler.
   *
   * @param game Current game instance
   */
  public AchievementHandler(Main game) {
    this.game = game;
    this.sprintUsed = false;
    this.noNegativeEvents = true;
  }

  /** Updates slow and steady Achievement when player has sprinted. */
  public void playerSprinted() {
    if (sprintUsed) {
      return;
    }
    this.sprintUsed = true;
  }

  /** Updates No Negative Events Achievement when player has triggered a negative Event. */
  public void triggerNegative() {
    if (!noNegativeEvents) {
      return;
    }
    noNegativeEvents = false;
  }

  /** Update longBoiDone Achievement when longboi is found. */
  public void triggerDuckFound() {
    if (duckFound) {
      return;
    }
    this.duckFound = true;
  }

  public boolean isSlowAndSteadyEarned() {
    return !sprintUsed;
  }

  public boolean isAllEventsTriggeredEarned() {
    return game.getEventsCounter().isAllEventsTriggered();
  }

  public boolean isNoNegativeEventsEarned() {
    return noNegativeEvents;
  }

  public boolean isFastEscapeEarned() {
    return !(game.getTimerManager().getTimeLeft() < 150);
  }

  public boolean isDuckFound() {
    return duckFound;
  }
}
