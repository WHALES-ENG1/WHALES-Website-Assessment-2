package com.mathochist.mazegame.userinterface;

import com.badlogic.gdx.Gdx;
import com.badlogic.gdx.graphics.OrthographicCamera;
import com.badlogic.gdx.graphics.Texture;
import com.badlogic.gdx.graphics.g2d.BitmapFont;
import com.badlogic.gdx.graphics.g2d.SpriteBatch;
import com.badlogic.gdx.graphics.g2d.TextureAtlas;
import com.badlogic.gdx.graphics.glutils.ShapeRenderer;
import com.badlogic.gdx.utils.viewport.FitViewport;
import com.badlogic.gdx.utils.viewport.Viewport;
import com.mathochist.mazegame.Main;
import com.mathochist.mazegame.userinterface.speechtext.SpeechBubbleManager;

/** Displays the HUD of the game over the screens. */
public class Hud {

  private static final float GAME_WIDTH = 1280;
  private static final float GAME_HEIGHT = 720;
  private final Main game;
  private final OrthographicCamera hudCamera;
  private final Viewport viewport;
  private final SpriteBatch hudBatch;
  private final BitmapFont smallFont;
  private final BitmapFont mainFont;
  private final BitmapFont vlargeFont;
  private final TextureAtlas gameUiAtlas;
  private final SpeechBubbleManager speechBubbleManager;

  /**
   * Displays using the games spriteBatch and sets the variables needed to display HUD.
   *
   * @param game current instance of game.
   */
  public Hud(Main game) {
    this.game = game;
    this.hudBatch = game.getSpriteBatch();
    hudCamera = new OrthographicCamera();
    viewport = new FitViewport(GAME_WIDTH, GAME_HEIGHT, hudCamera);
    hudCamera.position.set(GAME_WIDTH / 2.0f, GAME_HEIGHT / 2.0f, 0f);
    hudCamera.update();

    smallFont =
        new BitmapFont(
            Gdx.files.internal("fonts/game_font.fnt"),
            Gdx.files.internal("fonts/game_font.png"),
            false);
    smallFont.getData().setScale(0.5f);

    mainFont =
        new BitmapFont(
            Gdx.files.internal("fonts/game_font.fnt"),
            Gdx.files.internal("fonts/game_font.png"),
            false);
    mainFont.getData().setScale(1.0f);

    vlargeFont = new BitmapFont(Gdx.files.internal("fonts/vlarge_game_font.fnt"));

    gameUiAtlas = new TextureAtlas(Gdx.files.internal("game_ui/game_ui.atlas"));

    speechBubbleManager = new SpeechBubbleManager(this);
  }

  public SpriteBatch getHudBatch() {
    return hudBatch;
  }

  public OrthographicCamera getHudCamera() {
    return hudCamera;
  }

  public TextureAtlas getGameUiAtlas() {
    return gameUiAtlas;
  }

  public BitmapFont getSmallFont() {
    return smallFont;
  }

  public BitmapFont getMainFont() {
    return mainFont;
  }

  public BitmapFont getVlargeFont() {
    return vlargeFont;
  }

  public ShapeRenderer getShapeRenderer() {
    return game.getShapeRenderer();
  }

  /** disposes assets not in use. */
  public void dispose() {
    smallFont.dispose();
    mainFont.dispose();
    vlargeFont.dispose();
    gameUiAtlas.dispose();
    speechBubbleManager.dispose();
  }

  /**
   * Render HUD.
   */
  public void render() {
    viewport.apply();
    hudBatch.setProjectionMatrix(viewport.getCamera().combined);
    hudBatch.begin();

    // draw fps in top left corner
    smallFont.draw(
        hudBatch, "FPS: " + Gdx.graphics.getFramesPerSecond(), 2, hudCamera.viewportHeight - 2);

    // draw Pause instruction in top left corner
    hudBatch.draw(
        new Texture("maps/spare_assets_raw/keys/P.png"),
        2f,
        hudCamera.viewportHeight - 50,
        34f,
        32f);
    smallFont.draw(hudBatch, " -Pause", 35, hudCamera.viewportHeight - 30);
    // draw Achievement instruction
    hudBatch.draw(
        new Texture("maps/spare_assets_raw/keys/TAB.png"),
        2f,
        hudCamera.viewportHeight - 80,
        60f,
        32f);
    smallFont.draw(hudBatch, " -Achievements", 60, hudCamera.viewportHeight - 60);
    // draw Interact instruction
    hudBatch.draw(
        new Texture("maps/spare_assets_raw/keys/E.png"),
        2f,
        hudCamera.viewportHeight - 110,
        34f,
        32f);
    smallFont.draw(hudBatch, " -Interact", 35, hudCamera.viewportHeight - 90);
    // draw Sprint instruction
    hudBatch.draw(
        new Texture("maps/spare_assets_raw/keys/SHIFT.png"),
        2f,
        hudCamera.viewportHeight - 140,
        60f,
        32f);
    smallFont.draw(hudBatch, " -Sprint", 60, hudCamera.viewportHeight - 120);
    // draw time elapsed in top right corner
    game.getTimerManager().renderToHud(mainFont, hudBatch, hudCamera);
    game.getScoreCounter().renderToHud(mainFont, hudBatch, hudCamera);
    game.getPlayerInventory().render(hudBatch, gameUiAtlas);
    game.getEventsCounter().renderEventsCounterToHud(hudBatch, hudCamera, smallFont);

    speechBubbleManager.render();
    hudBatch.end();
  }

  public SpeechBubbleManager getSpeechBubbleManager() {
    return speechBubbleManager;
  }

  /** renders pause overlay. */
  public void renderPauseScreen() {
    viewport.apply();
    hudBatch.setProjectionMatrix(viewport.getCamera().combined);
    hudBatch.begin();

    float centerX = GAME_WIDTH / 2.0f;
    float centerY = GAME_HEIGHT / 2.0f;

    vlargeFont.draw(hudBatch, "Game Paused", centerX - 350, centerY + 100);
    mainFont.draw(hudBatch, "Press ESC or P to Resume", centerX - 150, centerY - 10);
    hudBatch.end();
  }

  /**
   * Resizes HUD when called.
   *
   * @param width New width.
   * @param height New height.
   */
  public void resize(int width, int height) {
    viewport.update(width, height, true);
  }
}
