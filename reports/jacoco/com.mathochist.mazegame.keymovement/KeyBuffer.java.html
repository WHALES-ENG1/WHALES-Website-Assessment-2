package com.mathochist.mazegame.keymovement;

import java.util.ArrayList;

/** A buffer to store key presses and camera movement. */
public class KeyBuffer {

  private final float[] cameraMove = {0, 0};
  private final ArrayList<Integer> keysPressed = new ArrayList<>();

  /** Create Key buffer. */
  public KeyBuffer() {}

  private void normalize() {
    if (cameraMove[0] != 0 || cameraMove[1] != 0) {
      float length =
          (float) Math.sqrt(cameraMove[0] * cameraMove[0] + cameraMove[1] * cameraMove[1]);
      cameraMove[0] /= length;
      cameraMove[1] /= length;
    }
  }

  /**
   * Return Camera move and normalize.
   *
   * @return cameraMove
   */
  public float[] getCameraMove() {
    normalize();
    return cameraMove;
  }

  public void setCameraX(float x) {
    cameraMove[0] = x;
  }

  public void setCameraY(float y) {
    cameraMove[1] = y;
  }

  /**
   * Move Camera by X.
   *
   * @param x to move by
   */
  public void addCameraX(float x) {
    cameraMove[0] += x;
  }

  /**
   * Move Camera by Y.
   *
   * @param y to move by
   */
  public void addCameraY(float y) {
    cameraMove[1] += y;
  }

  /** set camera x and y to 0. */
  public void clear() {
    cameraMove[0] = 0;
    cameraMove[1] = 0;
  }

  public ArrayList<Integer> getKeysPressed() {
    return keysPressed;
  }

  /**
   * add Pressed key to keysPressed.
   *
   * @param keycode keycode of pressed key
   */
  public void addKeyPressed(int keycode) {
    if (!keysPressed.contains(keycode)) {
      keysPressed.add(keycode);
    }
  }

  /**
   * remove Pressed Key.
   *
   * @param keycode keycode to remove
   */
  public void removeKeyPressed(int keycode) {
    keysPressed.remove((Integer) keycode);
  }

  /** Move Camera dependent on Key. */
  public void updateCameraMovementFromKeys() {
    clear();
    for (int keycode : keysPressed) {
      if (keycode == KeyBinds.MOVE_UP || keycode == KeyBinds.MOVE_UP_ALT) {
        addCameraY(1);
      }
      if (keycode == KeyBinds.MOVE_DOWN || keycode == KeyBinds.MOVE_DOWN_ALT) {
        addCameraY(-1);
      }
      if (keycode == KeyBinds.MOVE_LEFT || keycode == KeyBinds.MOVE_LEFT_ALT) {
        addCameraX(-1);
      }
      if (keycode == KeyBinds.MOVE_RIGHT || keycode == KeyBinds.MOVE_RIGHT_ALT) {
        addCameraX(1);
      }
    }
  }

  /**
   * Has the key just been pressed.
   *
   * @param keycode keycode to check
   * @return bool
   */
  public boolean isKeyPressed(int keycode) {
    return keysPressed.contains(keycode);
  }
}
