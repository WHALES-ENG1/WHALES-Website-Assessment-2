package com.mathochist.mazegame.entity.playerinventory;

/** Define Object, its name in atlas and if multiple can be attained. */
public enum InventoryObject {
  GLASSES(false, "glasses"),
  TORCH(false, "torch"),
  KEYCARD(false, "keycard"),
  ENERGY_DRINK(false, "energy_drink"),
  CHAMOMILE_TEA(false, "chamomile_tea"),
  BASEMENT_KEY(false, "basement_key"),
  FLASHLIGHT(true, "flashlight"),
  BATTERY(true, "battery"),
  MAP(false, "map"),
  MEDKIT(true, "medkit"),
  LONG_BOI(false, "long_boi");

  private final boolean allowMultiple;
  private final String objectName;

  InventoryObject(boolean allowMultiple, String objectName) {
    this.objectName = objectName;
    this.allowMultiple = allowMultiple;
  }

  /**
   * if object exists return object.
   *
   * @param name name of object
   * @return object
   */
  public static InventoryObject fromString(String name) {
    for (InventoryObject obj : InventoryObject.values()) {
      if (obj.getObjectName().equalsIgnoreCase(name)) {
        return obj;
      }
    }
    return null;
  }

  /**
   * Can you have multiple of this object.
   *
   * @return boolean
   */
  public boolean allowsMultiple() {
    return allowMultiple;
  }

  public String getObjectName() {
    return objectName;
  }
}
