package com.mathochist.mazegame.entity.playerinventory;

import com.badlogic.gdx.graphics.g2d.SpriteBatch;
import com.badlogic.gdx.graphics.g2d.TextureAtlas;
import java.util.ArrayList;

/** Class used to represent the objects the player holds and has interacted with. */
public class Inventory {

  private final ArrayList<InventoryObject> items;

  /** Create Inventory as a list. */
  public Inventory() {
    items = new ArrayList<>();
  }

  /**
   * Add item to the inventory.
   *
   * @param item item to add
   * @return has the item been added
   */
  public boolean addItem(InventoryObject item) {
    if (hasItem(item) && !item.allowsMultiple()) {
      return false;
    }
    items.add(item);
    return true;
  }

  /**
   * does the inventory contain item.
   *
   * @param item item to check
   * @return bool
   */
  public boolean hasItem(InventoryObject item) {
    return items.contains(item);
  }

  /** clear inventory. */
  public void clear() {
    items.clear();
  }

  /**
   * Render inventory items.
   *
   * @param batch SpriteBatch
   * @param gameUiAtlas Atlas that contains the inventory items
   */
  public void render(SpriteBatch batch, TextureAtlas gameUiAtlas) {
    int startX = 10;
    int startY = 10;
    int itemSize = 32;
    for (int i = 0; i < items.size(); i++) {
      InventoryObject item = items.get(i);
      batch.draw(
          gameUiAtlas.findRegion(item.getObjectName()),
          startX + i * (itemSize + 5),
          startY,
          itemSize,
          itemSize);
    }
  }
}
