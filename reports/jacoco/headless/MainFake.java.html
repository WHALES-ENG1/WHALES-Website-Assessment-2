package headless;

import static org.mockito.Mockito.when;

import com.badlogic.gdx.graphics.g2d.SpriteBatch;

import com.mathochist.mazegame.Main;
import com.mathochist.mazegame.entity.playerinventory.Inventory;

import com.mathochist.mazegame.userinterface.ScoreCounter;
import com.mathochist.mazegame.userinterface.timers.TimerManager;
import org.mockito.Mockito;

public class MainFake extends Main {
  private final SpriteBatch spriteBatch;
  private final ShapeRenderer shapeRenderer;
  private TimerManager timerManager;
  private Inventory playerInventory;
  private EventsCounter eventsCounter;
  private AchievementHandler achievementHandler;
  private ScoreCounter scoreCounter;

  public MainFake() {
    // Setup mock SpriteBatch & ShapeRenderer
    this.spriteBatch = Mockito.mock(SpriteBatch.class);
    when(this.spriteBatch.getTransformMatrix()).thenReturn(new Matrix4());
    when(this.spriteBatch.getProjectionMatrix()).thenReturn(new Matrix4());
    this.shapeRenderer = Mockito.mock(ShapeRenderer.class);
  }

  @Override

    this.timerManager = new TimerManager();

    this.scoreCounter = new ScoreCounter(this);
  }

  @Override

    return timerManager;
  }

  }

  }
