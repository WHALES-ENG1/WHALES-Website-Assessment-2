package headless;

import com.badlogic.gdx.backends.headless.HeadlessApplication;
import com.badlogic.gdx.backends.headless.HeadlessApplicationConfiguration;

/**
 * Launches the headless application. Can be converted into a utilities project or a server
 * application.
 *
 * @author Antonio Garcia-Dominguez
 * @author Samuel Knight
 */
public class HeadlessLauncher {
  protected static MainFake main;

  public static void main(String[] args) {
    createApplication();
  }

  private static void createApplication() {
    main = new MainFake();
    new HeadlessApplication(main, getDefaultConfiguration());
  }

  private static HeadlessApplicationConfiguration getDefaultConfiguration() {
    HeadlessApplicationConfiguration configuration = new HeadlessApplicationConfiguration();
    configuration.updatesPerSecond = -1; // When this value is negative, render() is never called.
    return configuration;
  }

  public static MainFake getMain() {
    return main;
  }
}
